local SyncREQ =
    (syn and syn.request)
    or (http and http.request)
    or http_request
    or request

local function SyncAPI(Method, Path, Body)
    if not V4Sync.Enabled or SyncRole == "UNKNOWN" or not SyncREQ then
        return nil, nil, "sync_unavailable"
    end

    local ok, response = pcall(function()
        return SyncREQ({
            Url = V4Sync.API_URL .. Path,
            Method = Method,
            Headers = {
                ["Content-Type"] = "application/json",
                ["x-api-key"] = V4Sync.API_KEY
            },
            Body = Body and HttpService:JSONEncode(Body) or nil
        })
    end)

    if not ok then
        warn("[NUVEM V4 SITE DEBUG] HTTP ERROR | method=" .. tostring(Method) .. " | path=" .. tostring(Path) .. " | err=" .. tostring(response))
        return nil, nil, tostring(response)
    end

    local statusCode = tonumber(response.StatusCode) or 0
    local bodyText = tostring(response.Body or "")

    print("[NUVEM V4 SITE DEBUG] HTTP | " .. tostring(Method) .. " " .. tostring(Path) .. " | status=" .. tostring(statusCode))

    local data
    local decodeOK, decodeErr = pcall(function()
        data = HttpService:JSONDecode(bodyText)
    end)

    if not decodeOK then
        warn("[NUVEM V4 SITE DEBUG] JSON DECODE ERROR | " .. tostring(decodeErr))
        print("[NUVEM V4 SITE DEBUG] BODY=" .. bodyText)
    elseif type(data) == "table" then
        print(
            "[NUVEM V4 SITE DEBUG] RESPONSE | command=" .. tostring(data.command)
            .. " | phase=" .. tostring(data.phase)
            .. " | state=" .. tostring(data.state)
            .. " | cycleId=" .. tostring(data.cycleId)
        )
    else
        print("[NUVEM V4 SITE DEBUG] RESPONSE TYPE=" .. typeof(data))
    end

    return statusCode, data, response.StatusMessage
end

local function SyncReport(State, Extra)
    if SyncRole == "UNKNOWN" or not SyncREQ then
        return nil
    end

    local Root = GetRoot and GetRoot() or nil
    local Race = "Unknown"
    pcall(function()
        if typeof(GetCurrentRace) == "function" then
            Race = GetCurrentRace() or "Unknown"
        end
    end)

    if Race == "Unknown" then
        pcall(function()
            local Data = Player:FindFirstChild("Data")
            local RaceValue = Data and Data:FindFirstChild("Race")
            if RaceValue and RaceValue.Value then
                Race = tostring(RaceValue.Value)
            end
        end)
    end

    local Distance = 99999
    local AtDoor = false
    pcall(function()
        local Door = GetTrialDoorCFrame()
        if Root and Door then
            Distance = (Root.Position - Door.Position).Magnitude
            AtDoor = Distance <= 15
        end
    end)

    local Body = {
        account = Player.Name,
        jobId = game.JobId,
        race = Race,
        state = State or SyncState,
        distance = Distance,
        atDoor = AtDoor,
        alive = Root ~= nil,
        cycleId = SyncCycleId,
        extra = Extra
    }

    local Code, Data = SyncAPI("POST", "/v4/state", Body)

    if Code == 200 and Data then
        SyncRegistered = true
        SyncState = State or SyncState
        SyncPhase = tostring(Data.phase or SyncPhase)
        SyncCycleId = tonumber(Data.cycleId) or SyncCycleId
        SyncCommand = tostring(Data.command or "WAIT")
        SyncLastReport = tick()
        return Data
    end

    return nil
end

local function SyncRegister()
    if SyncRole == "UNKNOWN" or not SyncREQ then
        return nil
    end

    local Race = "Unknown"
    pcall(function()
        if typeof(GetCurrentRace) == "function" then
            Race = GetCurrentRace() or "Unknown"
        end
    end)

    if Race == "Unknown" then
        pcall(function()
            local Data = Player:FindFirstChild("Data")
            local RaceValue = Data and Data:FindFirstChild("Race")
            if RaceValue and RaceValue.Value then
                Race = tostring(RaceValue.Value)
            end
        end)
    end

    local Code, Data = SyncAPI("POST", "/v4/register", {
        account = Player.Name,
        jobId = game.JobId,
        race = Race,
        state = "WAITING"
    })

    if Code == 200 and Data then
        SyncRegistered = true
        SyncPhase = tostring(Data.phase or "WAITING_READY")
        SyncCycleId = tonumber(Data.cycleId) or 1
        SyncCommand = tostring(Data.command or "WAIT")
        print("[NUVEM V4 SYNC] Registrado | role=", SyncRole, "| phase=", SyncPhase, "| cycle=", SyncCycleId)

        -- CORREÇÃO DE CICLO PRESO:
        -- Se a conta MAIN foi reiniciada, limpa automaticamente um ciclo
        -- antigo que possa ter ficado em WAITING_TRIAL_RESULT.
        -- As AUX não fazem esse reset para não derrubar o grupo ao reiniciar
        -- apenas uma conta auxiliar.
        if SyncRole == "MAIN" then
            local ResetCode, ResetData = SyncAPI("POST", "/v4/reset", {
                account = Player.Name,
                jobId = game.JobId,
                reason = "main_startup"
            })

            if ResetCode == 200 and ResetData then
                SyncRegistered = true
                SyncState = "WAITING"
                SyncPhase = tostring(ResetData.phase or "WAITING_READY")
                SyncCycleId = tonumber(ResetData.cycleId) or SyncCycleId
                SyncCommand = tostring(ResetData.command or "WAIT")
                SyncStartIssued = false
                SyncTrialReported = false
                SyncResetRequested = false
                SyncPostResetRoute = false
                SyncWaitingForGroup = false
                V3ActivatedForTrial = false
                print("[NUVEM V4 SYNC] Ciclo antigo limpo no startup da MAIN | phase=", SyncPhase, "| cycle=", SyncCycleId)
                return ResetData
            else
                warn("[NUVEM V4 SYNC] Não foi possível limpar o ciclo antigo no startup da MAIN.")
            end
        end

        return Data
    end

    return nil
end

local function SyncPoll()
    if SyncRole == "UNKNOWN" then
        print("[NUVEM V4 SITE DEBUG] POLL BLOQUEADO | role=UNKNOWN | account=" .. tostring(Player.Name))
        return nil
    end

    if not SyncREQ then
        warn("[NUVEM V4 SITE DEBUG] POLL BLOQUEADO | request function não encontrada")
        return nil
    end

    local Path = "/v4/status?account=" .. HttpService:UrlEncode(Player.Name)

    print(
        "[NUVEM V4 SITE DEBUG] POLL -> "
        .. tostring(Player.Name)
        .. " | role=" .. tostring(SyncRole)
    )

    local Code, Data, Message = SyncAPI("GET", Path)

    print(
        "[NUVEM V4 SITE DEBUG] POLL RESULT | code="
        .. tostring(Code)
        .. " | message=" .. tostring(Message)
        .. " | data=" .. tostring(Data ~= nil)
    )

    if Code == 200 and Data then
        SyncRegistered = true
        SyncPhase = tostring(Data.phase or SyncPhase)
        SyncCycleId = tonumber(Data.cycleId) or SyncCycleId
        SyncCommand = tostring(Data.command or "WAIT")

        print(
            "[NUVEM V4 SITE DEBUG] ESTADO APLICADO | command="
            .. tostring(SyncCommand)
            .. " | phase=" .. tostring(SyncPhase)
            .. " | cycle=" .. tostring(SyncCycleId)
        )

        return Data
    end

    warn("[NUVEM V4 SITE DEBUG] POLL FALHOU | nenhum comando foi aplicado")
    return nil
end

-- Cache opcional e totalmente protegido do progresso REAL já confirmado.
-- Se o executor não oferecer readfile/isfile/writefile, o script continua
-- normalmente sem depender dessas funções.
local ProgressCacheFile = "NuvemV4_Progress_" .. tostring(Player.UserId) .. ".txt"

local function LoadCachedV4Progress()
    local Success, Value = pcall(function()
        local Env = _G

        local GetGenEnv = rawget(_G, "getgenv")
        if type(GetGenEnv) == "function" then
            local Candidate = GetGenEnv()
            if type(Candidate) == "table" then
                Env = Candidate
            end
        end

        local IsFile = rawget(Env, "isfile")
        local ReadFile = rawget(Env, "readfile")

        if type(IsFile) ~= "function" or type(ReadFile) ~= "function" then
            return nil
        end

        if not IsFile(ProgressCacheFile) then
            return nil
        end

        local Content = ReadFile(ProgressCacheFile)
        local Number = tonumber(Content)

        if Number == nil then
            return nil
        end

        Number = math.floor(Number)

        if Number < 0 then
            Number = 0
        elseif Number > 10 then
            Number = 10
        end

        return Number
    end)

    if Success and type(Value) == "number" then
        return Value
    end

    return nil
end

local function SaveCachedV4Progress(Value)
    pcall(function()
        if type(Value) ~= "number" then
            return
        end

        Value = math.floor(Value)

        if Value < 0 then
            Value = 0
        elseif Value > 10 then
            Value = 10
        end

        local Env = _G

        local GetGenEnv = rawget(_G, "getgenv")
        if type(GetGenEnv) == "function" then
            local Candidate = GetGenEnv()
            if type(Candidate) == "table" then
                Env = Candidate
            end
        end

        local WriteFile = rawget(Env, "writefile")

        if type(WriteFile) == "function" then
            WriteFile(ProgressCacheFile, tostring(Value))
        end
    end)
end

end

-- Aviso visual no cliente quando o estado do Ancient One muda.
-- Não depende do Developer Console.
local function NotifyAncientOne(state, message)

    local key = tostring(state) .. "|" .. tostring(message)

    if LastAncientNotification == key then
        return
    end

    LastAncientNotification = key

    local title = "Ancient One"

    if state == "TRIAL" then
        title = "Ancient One: Ready For Trial"
    elseif state == "TRAINING" then
        title = "Ancient One: Treinamento"
    elseif state == "BUY_AVAILABLE" then
        title = "Ancient One: Upgrade disponível"
    elseif state == "COMPLETE" then
        title = "Ancient One: Concluído"
    end

    pcall(function()
        game:GetService("StarterGui"):SetCore(
            "SendNotification",
            {
                Title = title,
                Text = tostring(message),
                Duration = 6
            }
        )
    end)

    print("[ANCIENT ONE] " .. title .. " | " .. tostring(message))
end

--========================================================

--========================================================

local function EquipFightingStyle()

    if tick() - LastEquip < 1 then
        return
    end

    LastEquip = tick()

    local CurrentCharacter = Player.Character

    if not CurrentCharacter then
        return
    end

    local Humanoid =
        CurrentCharacter:FindFirstChildOfClass("Humanoid")

    if not Humanoid then
        return
    end

    local Equipped =
        CurrentCharacter:FindFirstChildOfClass("Tool")

    if Equipped and Equipped.ToolTip == "Melee" then
        return
    end

    local Backpack =
        Player:FindFirstChildOfClass("Backpack")

    if not Backpack then
        return
    end

    for _, Tool in ipairs(Backpack:GetChildren()) do

        if Tool:IsA("Tool")
            and Tool.ToolTip == "Melee"
        then

            pcall(function()
                Humanoid:EquipTool(Tool)
            end)

            return

        end

    end

end

local MAX_DISTANCE = 35
local MAX_PULL = 3      

local function MagnetNPCs()
    local Root = GetRoot()
    if not Root then return end

    local count = 0
    -- Aumentado de -5 para -8 studs abaixo do jogador
    local targetCFrame = Root.CFrame * CFrame.new(0, -8, 0)

    for _, NPC in ipairs(Enemies:GetChildren()) do
        if count >= MAX_PULL then break end

        if BonesNPCs[NPC.Name] and IsAlive(NPC) then
            local npcHrp = NPC:FindFirstChild("HumanoidRootPart")
            local humanoid = NPC:FindFirstChildOfClass("Humanoid")

            if npcHrp and humanoid then
                local distance = (npcHrp.Position - Root.Position).Magnitude

                if distance <= MAX_DISTANCE then
                    humanoid.WalkSpeed = 0
                    npcHrp.CanCollide = false
                    npcHrp.Size = Vector3.new(6, 6, 6)

                    npcHrp.Anchored = true
                    npcHrp.CFrame = targetCFrame

                    count = count + 1
                end
            end
        end
    end
end

local function StopMovement()

    Moving = false
    ShouldTween = false

    if CurrentTween then

        pcall(function()
            CurrentTween:Cancel()
        end)

        CurrentTween = nil

    end

    local Root = GetRoot()

    if Root then

        Root.AssemblyLinearVelocity = Vector3.zero
        Root.AssemblyAngularVelocity = Vector3.zero

    end

end

local function MoveTo(TargetCFrame, SpeedOverride)

    local Root = GetRoot()

    if not Root then
        return false
    end

    local Distance =
        (
            TargetCFrame.Position
            - Root.Position
        ).Magnitude

    if Distance <= Config.TargetDistance then

        StopMovement()

        Root.CFrame = TargetCFrame

        return true

    end

    if CurrentTween
        and Moving
        and CurrentTween.PlaybackState
            == Enum.PlaybackState.Playing
    then

        local CurrentDestination =
            MoveBlock.CFrame.Position

        if (
            CurrentDestination
            - TargetCFrame.Position
        ).Magnitude <= 20 then

            return false

        end

        pcall(function()
            CurrentTween:Cancel()
        end)

        CurrentTween = nil

    end

    MoveBlock.CFrame = Root.CFrame

    local MoveSpeed = tonumber(SpeedOverride) or Config.TravelSpeed

    local Duration =
        math.max(
            Distance / MoveSpeed,
            0.05
        )

    CurrentTween =
        TweenService:Create(
            MoveBlock,

            TweenInfo.new(
                Duration,
                Enum.EasingStyle.Linear,
                Enum.EasingDirection.Out
            ),

            {
                CFrame = TargetCFrame
            }
        )

    Moving = true
    ShouldTween = true

    CurrentTween:Play()

    return false

end

--========================================================
-- NAVEGAÇÃO BONES - MÉTODO IORGUTE
--========================================================
--
-- A lógica aqui segue o padrão observado no iorgute:
-- 1) escolhe UM Bone real como alvo;
-- 2) salva uma posição "Locked" desse alvo;
-- 3) usa essa posição como PosMon;
-- 4) traz outros Bones para a mesma região;
-- 5) mantém o jogador acima da posição bloqueada;
-- 6) não troca o alvo a cada ciclo, evitando subir/teleportar
--    para posições diferentes.
--
local TrainingLocked = false
local TrainingLockedPosition = nil
local TrainingTarget = nil

local function ReleaseTrainingNPCs()

    for _, NPC in ipairs(Enemies:GetChildren()) do

        if BonesNPCs[NPC.Name] then

            local Humanoid =
                NPC:FindFirstChildOfClass("Humanoid")

            local Root =
                NPC:FindFirstChild("HumanoidRootPart")

            if Humanoid then
                pcall(function()
                    Humanoid.WalkSpeed = 16
                    Humanoid.JumpPower = 50
                end)
            end

            if Root then
                pcall(function()
                    Root.CanCollide = true
                    Root.Anchored = false
                end)
            end

        end

    end

    TrainingLocked = false
    TrainingLockedPosition = nil
    TrainingTarget = nil

end

local function FindTrainingTarget()

    local Root = GetRoot()

    if not Root then
        return nil
    end

    local Closest
    local ClosestDistance = math.huge

    for _, NPC in ipairs(Enemies:GetChildren()) do

        if BonesNPCs[NPC.Name] and IsAlive(NPC) then

            local NPCRoot =
                NPC:FindFirstChild("HumanoidRootPart")

            if NPCRoot then

                local Distance =
                    (
                        NPCRoot.Position
                        - Root.Position
                    ).Magnitude

                if Distance < ClosestDistance then
                    ClosestDistance = Distance
                    Closest = NPC
                end

            end

        end

    end

    return Closest

end

local function BringTrainingNPCs()

    if not TrainingLocked
        or not TrainingLockedPosition
    then
        return
    end

    local Root = GetRoot()

    if not Root then
        return
    end

    local Count = 0

    -- Mesmo princípio do BringEnemy do iorgute:
    -- NPCs válidos próximos são colocados no PosMon.
    for _, NPC in ipairs(Enemies:GetChildren()) do

        if Count >= 3 then
            break
        end

        if BonesNPCs[NPC.Name]
            and IsAlive(NPC)
        then

            local NPCRoot =
                NPC:FindFirstChild("HumanoidRootPart")

            local Humanoid =
                NPC:FindFirstChildOfClass("Humanoid")

            if NPCRoot and Humanoid then

                local Distance =
                    (
                        NPCRoot.Position
                        - TrainingLockedPosition.Position
                    ).Magnitude

                if Distance <= 100 then

                    pcall(function()

                        Humanoid.WalkSpeed = 0
                        Humanoid.JumpPower = 0

                        NPCRoot.CanCollide = true
                        NPCRoot.Anchored = false

                        -- Pequenas separações para que os 3
                        -- não fiquem exatamente sobrepostos.
                        local Offset =
                            Vector3.new(
                                (Count - 1) * 3,
                                0,
                                0
                            )

                        NPCRoot.CFrame =
                            CFrame.new(
                                TrainingLockedPosition.Position
                                + Offset
                            )

                        NPCRoot.AssemblyLinearVelocity =
                            Vector3.zero

                        NPCRoot.AssemblyAngularVelocity =
                            Vector3.zero

                    end)

                    Count = Count + 1

                end

            end

        end

    end

    -- O jogador fica fixo acima do grupo.
    local PlayerPosition =
        TrainingLockedPosition.Position
        + Vector3.new(
            0,
            Config.FarmHeight,
            0
        )

    Root.CFrame =
        CFrame.new(
            PlayerPosition
        )

    Root.AssemblyLinearVelocity =
        Vector3.zero

    Root.AssemblyAngularVelocity =
        Vector3.zero

end

--========================================================
-- NAVEGAÇÃO BONES - BRINGENEMY / POSMON
--========================================================
--
-- Esta versão usa a mesma estrutura da source analisada:
--
-- 1) escolhe UM Bone como alvo;
-- 2) salva a posição ORIGINAL dele como PosMon;
-- 3) o jogador vai para cima dessa posição;
-- 4) BringBoneMob() é chamado continuamente;
-- 5) os outros Bones dentro do raio são trazidos para PosMon;
-- 6) o jogador não acompanha a posição dos NPCs.
--

local BoneFarmLocked = false
local BoneFarmCFrame = nil
local BoneFarmTarget = nil
local BoneFarmTargets = {}

local TrainingMoveStarted = false
local TrainingAtBones = false

local BONE_BRING_DISTANCE = 300
local BONE_MAX_TARGETS = 3

local function ClearBoneFarm()

    BoneFarmLocked = false
    BoneFarmCFrame = nil
    BoneFarmTarget = nil
    BoneFarmTargets = {}

end

local function GetBoneRoot(NPC)

    if not NPC then
        return nil
    end

    return NPC.PrimaryPart
        or NPC:FindFirstChild("HumanoidRootPart")

end

local function FindFirstBone()

    local Root = GetRoot()

    if not Root then
        return nil
    end

    local Closest = nil
    local ClosestDistance = math.huge

    for _, NPC in ipairs(Enemies:GetChildren()) do

        if BonesNPCs[NPC.Name]
            and IsAlive(NPC)
        then

            local NPCRoot =
                GetBoneRoot(NPC)

            if NPCRoot then

                local Distance =
                    (
                        NPCRoot.Position
                        - Root.Position
                    ).Magnitude

                if Distance < ClosestDistance then

                    ClosestDistance = Distance
                    Closest = NPC

                end
            end
        end
    end

    return Closest

end

local function BringBoneMob()

    if not BoneFarmLocked
        or not BoneFarmCFrame
    then
        return
    end

    local PosMon =
        BoneFarmCFrame.Position

    local Count = 0
    local Targets = {}

    --====================================================
    -- BRING ENEMY
    --====================================================
    --
    -- Diferente das versões anteriores:
    -- NÃO usa a posição atual do jogador para decidir
    -- se o NPC pode ser puxado.
    --
    -- PosMon é a posição salva do primeiro Bone.
    -- O raio de 300 permite juntar os Bones da área.
    --

    for _, NPC in ipairs(Enemies:GetChildren()) do

        if Count >= BONE_MAX_TARGETS then
            break
        end

        if BonesNPCs[NPC.Name]
            and IsAlive(NPC)
        then

            local NPCRoot =
                GetBoneRoot(NPC)

            local Humanoid =
                NPC:FindFirstChildOfClass(
                    "Humanoid"
                )

            if NPCRoot and Humanoid then

                local Distance =
                    (
                        NPCRoot.Position
                        - PosMon
                    ).Magnitude

                if Distance <= BONE_BRING_DISTANCE then

                    pcall(function()

                        -- Mesmo conceito do BringEnemy:
                        -- joga o PrimaryPart diretamente no PosMon.
                        NPCRoot.CFrame =
                            CFrame.new(
                                PosMon
                            )

                        NPCRoot.CanCollide =
                            false

                        NPCRoot.AssemblyLinearVelocity =
                            Vector3.zero

                        NPCRoot.AssemblyAngularVelocity =
                            Vector3.zero

                        Humanoid.WalkSpeed = 0
                        Humanoid.JumpPower = 0

                    end)

                    table.insert(
                        Targets,
                        NPC
                    )

                    Count =
                        Count + 1

                end
            end
        end
    end

    BoneFarmTargets =
        Targets

    -- Mantém o raio de simulação do cliente alto,
    -- como no padrão analisado.
    pcall(function()
        Player.SimulationRadius =
            math.huge
    end)

end

local function NavigateBones()

    local Root = GetRoot()

    if not Root then
        return
    end

    --====================================================
    -- FARM JÁ TRAVADO
    --====================================================

    if BoneFarmLocked
        and BoneFarmCFrame
    then

        -- O jogador fica FIXO no ponto salvo.
        Root.CFrame =
            BoneFarmCFrame
            + Vector3.new(
                0,
                Config.FarmHeight,
                0
            )

        Root.AssemblyLinearVelocity =
            Vector3.zero

        Root.AssemblyAngularVelocity =
            Vector3.zero

        -- Se o alvo morreu, escolhe outro Bone.
        -- A posição do jogador NÃO muda para acompanhar
        -- o novo NPC enquanto o grupo continua.
        if not BoneFarmTarget
            or not BoneFarmTarget.Parent
            or not IsAlive(BoneFarmTarget)
        then

            local NewTarget = nil
            local NewDistance = math.huge

            for _, NPC in ipairs(
                Enemies:GetChildren()
            ) do

                if BonesNPCs[NPC.Name]
                    and IsAlive(NPC)
                then

                    local NPCRoot =
                        GetBoneRoot(NPC)

                    if NPCRoot then

                        local Distance =
                            (
                                NPCRoot.Position
                                - BoneFarmCFrame.Position
                            ).Magnitude

                        if Distance < NewDistance
                            and Distance <= BONE_BRING_DISTANCE
                        then

                            NewDistance =
                                Distance

                            NewTarget =
                                NPC

                        end
                    end
                end
            end

            BoneFarmTarget =
                NewTarget

        end

        if BoneFarmTarget then

            -- BringEnemy roda a cada ciclo.
            BringBoneMob()

            return

        end

        ClearBoneFarm()
        CurrentTarget = nil

        return

    end

    --====================================================
    -- ESCOLHE O PRIMEIRO BONE
    --====================================================

    local FirstTarget =
        FindFirstBone()

    if not FirstTarget then

        StopMovement()
        return

    end

    local FirstRoot =
        GetBoneRoot(
            FirstTarget
        )

    if not FirstRoot then
        return
    end

    CurrentTarget =
        FirstTarget

    BoneFarmTarget =
        FirstTarget

    --====================================================
    -- POSMON = POSIÇÃO ORIGINAL DO PRIMEIRO NPC
    --====================================================

    local PosMon =
        CFrame.new(
            FirstRoot.Position
        )

    local TargetCFrame =
        PosMon
        + Vector3.new(
            0,
            Config.FarmHeight,
            0
        )

    local Distance =
        (
            Root.Position
            - TargetCFrame.Position
        ).Magnitude

    if Distance > Config.TargetDistance then

        MoveTo(TargetCFrame)
        return

    end

    if CurrentTween then

        pcall(function()
            CurrentTween:Cancel()
        end)

        CurrentTween = nil

    end

    Moving = false
    ShouldTween = false

    -- Salva o PosMon UMA vez.
    BoneFarmCFrame =
        PosMon

    BoneFarmLocked =
        true

    -- Fica acima do PosMon.
    Root.CFrame =
        BoneFarmCFrame
        + Vector3.new(
            0,
            Config.FarmHeight,
            0
        )

    Root.AssemblyLinearVelocity =
        Vector3.zero

    Root.AssemblyAngularVelocity =
        Vector3.zero

    -- Primeiro BringEnemy.
    BringBoneMob()

end

local function ActivateBuso()

    if not Config.AutoBuso then
        return
    end

    if not CommF then
        return
    end

    if tick() - LastBuso < 2 then
        return
    end

    local CurrentCharacter =
        Player.Character

    if not CurrentCharacter then
        return
    end

    if CurrentCharacter:FindFirstChild("HasBuso") then
        return
    end

    LastBuso = tick()

    pcall(function()
        CommF:InvokeServer("Buso")
    end)

end

local function SetStableV4Progress(Value, Source)
    if typeof(Value) ~= "number" then
        return StableV4Progress
    end

    Value = math.clamp(math.floor(Value), 0, 10)

    -- O progresso do ciclo só pode avançar.
    -- Isso impede que um retorno transitório do Ancient One
    -- faça a UI voltar de 5/10 para 0/10.
    if Value < StableV4Progress then
        print(
            "[NUVEM V4] Progresso regressivo ignorado:",
            tostring(Value),
            "->",
            tostring(StableV4Progress),
            "| Fonte:",
            tostring(Source or "desconhecida")
        )
        V4Progress = StableV4Progress
        return StableV4Progress
    end

    StableV4Progress = Value
    V4Progress = Value

    -- Persiste somente valores aceitos como progresso REAL.
    -- Falha de filesystem nunca pode quebrar o script principal.
    SaveCachedV4Progress(Value)

    return Value
end

local function GetProgressText()
    return tostring(StableV4Progress) .. "/10"
end

local function ReadAncientOneDialogText()
    local PlayerGui = Player:FindFirstChild("PlayerGui")
    if not PlayerGui then
        return ""
    end

    local Texts = {}

    for _, Object in ipairs(PlayerGui:GetDescendants()) do
        if Object:IsA("TextLabel")
            or Object:IsA("TextButton")
            or Object:IsA("TextBox")
        then
            local IsVisible = true
            pcall(function()
                IsVisible = Object.Visible == true
            end)

            local Text = tostring(Object.Text or "")

            if IsVisible and Text ~= "" then
                local Lower = string.lower(Text)

                if string.find(Lower, "trial")
                    or string.find(Lower, "train")
                    or string.find(Lower, "ready")
                then
                    table.insert(Texts, Text)
                end
            end
        end
    end

    return table.concat(Texts, " | ")
end

local function ProbeAncientOneDialog()
    if AncientOneDialogBusy then
        return AncientOneDialogState
    end

    AncientOneDialogBusy = true
    AncientOneDialogState = nil

    local Success, Result = pcall(function()
        local NPC = GetAncientOneModel()

        if not NPC then
            warn("[NUVEM V4] Ancient One não encontrado para diagnóstico de diálogo.")
            return nil
        end

        print("[NUVEM V4] Ancient One encontrado:", NPC:GetFullName())

        -- Tenta abrir a interação local somente se o executor oferecer
        -- uma das funções de interação. Se não houver, o servidor continua
        -- sendo a fonte principal de verdade.
        local Prompt = NPC:FindFirstChildWhichIsA("ProximityPrompt", true)
        if Prompt and typeof(fireproximityprompt) == "function" then
            pcall(function()
                fireproximityprompt(Prompt, math.huge)
            end)
        else
            local Click = NPC:FindFirstChildWhichIsA("ClickDetector", true)
            if Click and typeof(fireclickdetector) == "function" then
                pcall(function()
                    fireclickdetector(Click)
                end)
            end
        end

        local Started = tick()

        while tick() - Started < 2.5 do
            local Text = ReadAncientOneDialogText()
            local Lower = string.lower(Text)

            if (string.find(Lower, "ready")
                    and string.find(Lower, "trial"))
                or string.find(Lower, "next trial")
            then
                print("[NUVEM V4] Diálogo detectado: READY / TRIAL")
                print("[NUVEM V4] Texto:", Text)
                return "READY_FOR_TRIAL"
            end

            if (string.find(Lower, "need")
                    and string.find(Lower, "train"))
                or string.find(Lower, "need to train")
                or string.find(Lower, "needs training")
            then
                print("[NUVEM V4] Diálogo detectado: PRECISA TREINAR")
                print("[NUVEM V4] Texto:", Text)
                return "NEED_TRAINING"
            end

            if string.find(Lower, "train")
                or string.find(Lower, "training")
            then
                print("[NUVEM V4] Diálogo detectado: TREINAMENTO")
                print("[NUVEM V4] Texto:", Text)
                return "NEED_TRAINING"
            end

            task.wait(0.15)
        end

        print("[NUVEM V4] Diálogo do Ancient One não reconhecido.")
        return "UNKNOWN"
    end)

    if Success then
        AncientOneDialogState = Result
    else
        warn("[NUVEM V4] Falha no diagnóstico do diálogo:", Result)
        AncientOneDialogState = "UNKNOWN"
    end

    AncientOneDialogBusy = false
    return AncientOneDialogState
end

local AncientOnePosition = CFrame.new(
    28981.552734375,
    14888.4267578125,
    -120.245849609375
)

local function RunAncientOneProbe()
    if not AncientOneProbeRequired or AncientOneProbeDone then
        return true
    end

    local Root = GetRoot()
    if not Root then
        return false
    end

    local Distance =
        (Root.Position - AncientOnePosition.Position).Magnitude

    if Distance > Config.TargetDistance then
        UpdateStatusUI(
            "VERIFICANDO",
            "Conferindo fala do Ancient One...",
            GetProgressText()
        )

        MoveTo(AncientOnePosition)
        return false
    end

    StopMovement()

    local Result = ProbeAncientOneDialog()

    if Result == "READY_FOR_TRIAL" then
        AncientOneDialogState = Result
        AncientOneProbeDone = true
        AncientOneProbeRequired = false

        print("[NUVEM V4] Ancient One confirmou READY FOR TRIAL pelo diálogo.")

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            "Ancient One confirmou — indo para a porta...",
            GetProgressText()
        )

        return true
    end

    if Result == "NEED_TRAINING" then
        AncientOneDialogState = Result
        AncientOneProbeDone = true
        AncientOneProbeRequired = false
        V4State = "TRAINING"
        V4CanFarmBones = true

        print("[NUVEM V4] Ancient One informou TREINAMENTO pelo diálogo.")

        UpdateStatusUI(
            "PRECISA TREINAR",
            "Ancient One informou que precisa treinar.",
            GetProgressText()
        )

        return true
    end

    -- Sem diálogo reconhecido, não vamos adivinhar que está pronto.
    -- O fluxo volta ao Check do servidor.
    print("[NUVEM V4] Diálogo não reconhecido; mantendo Check como fonte principal.")
    AncientOneProbeDone = true
    AncientOneProbeRequired = false
    return true
end

local function GoToTempleForTrial()

    if TempleReady or TempleEntering then
        return
    end

    if not CommF then

        warn(
            "[NUVEM V4] CommF_ indisponível para entrar no Temple."
        )

        return

    end

    TempleEntering = true

    StopMovement()
    CurrentTarget = nil

    UpdateStatusUI(
        "PRONTO PARA TRIAL",
        "Preparando Temple of Time...",
        GetProgressText()
    )

    local Success, Error = pcall(function()

        local Map =
            Workspace:WaitForChild("Map")

        if not Map:FindFirstChild("Temple of Time") then

            local MapStash =
                ReplicatedStorage:FindFirstChild(
                    "MapStash"
                )

            if MapStash then

                local Temple =
                    MapStash:FindFirstChild(
                        "Temple of Time"
                    )

                if Temple then

                    Temple.Parent = Map

                    task.wait(0.5)

                    print(
                        "[NUVEM V4] Temple of Time carregado do MapStash."
                    )

                else

                    warn(
                        "[NUVEM V4] Temple of Time não encontrado no MapStash."
                    )

                end

            else

                warn(
                    "[NUVEM V4] MapStash não encontrado."
                )

            end

        end

        local Entrance = Vector3.new(
            28286.56,
            14933.21,
            103.34
        )

        print(
            "[NUVEM V4] Solicitando entrada no Temple of Time..."
        )

        CommF:InvokeServer(
            "requestEntrance",
            Entrance
        )

        task.wait(0.5)

        Character =
            Player.Character
            or Player.CharacterAdded:Wait()

        local Root =
            Character:WaitForChild(
                "HumanoidRootPart"
            )

        local HoldPosition =
            Root.Position

        print(
            "[NUVEM V4] Posição após teleporte:",
            tostring(HoldPosition)
        )

        local StartTime = tick()

        while
            tick() - StartTime < 3
            and Config.Enabled
        do

            if not Root
                or not Root.Parent
            then
                break
            end

            Root.CFrame =
                CFrame.new(HoldPosition)

            Root.AssemblyLinearVelocity =
                Vector3.zero

            Root.AssemblyAngularVelocity =
                Vector3.zero

            RunService.Heartbeat:Wait()

        end

        TempleReady = true
        TempleEntering = false
        TrialDoorReady = false

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            "Temple alcançado — indo para a porta da Trial...",
            GetProgressText()
        )

        print(
            "[NUVEM V4] Temple of Time alcançado."
        )

        print(
            "[NUVEM V4] Próximo passo: porta da Trial."
        )

    end)

    if not Success then

        TempleReady = false
        TempleEntering = false

        warn(
            "[NUVEM V4] Falha ao entrar no Temple of Time:",
            Error
        )

    end

end

local function GetCurrentRace()
    local Data = Player:FindFirstChild("Data")
    local Race = Data and Data:FindFirstChild("Race")

    if not Race then
        return nil
    end

    return tostring(Race.Value)
end

local function GetTrialLocation()
    local Race = GetCurrentRace()
    local LocationName = Race and RaceToTrial[Race]

    if not LocationName then
        return nil, nil, nil
    end

    local WorldOrigin = Workspace:FindFirstChild("_WorldOrigin")
    local Locations = WorldOrigin and WorldOrigin:FindFirstChild("Locations")

    if not Locations then
        return nil, Race, LocationName
    end

    local Location = Locations:FindFirstChild(LocationName)

    if not Location then
        return nil, Race, LocationName
    end

    return Location, Race, LocationName
end

local function GetTrialPosition(Object)
    if not Object then
        return nil
    end

    local Position

    pcall(function()
        if Object:IsA("BasePart") then
            Position = Object.Position
        elseif Object:IsA("Model") then
            Position = Object:GetPivot().Position
        end
    end)

    return Position
end

local function IsPlayerInsideExpectedTrial()
    local Root = GetRoot()

    if not Root then
        return false
    end

    local Location, Race, Trial = GetTrialLocation()

    if not Location then
        return false
    end

    local Position = GetTrialPosition(Location)

    if not Position then
        return false
    end

    local Radius = Location:GetAttribute("Radius")

    if typeof(Radius) ~= "number" then
        Radius = 1200
    end

    -- Radius muito grande pode pegar regiões vizinhas.
    Radius = math.min(Radius, 1500)

    local Distance = (Root.Position - Position).Magnitude

    return Distance <= Radius
end

local function IsRaidTimerVisible()

    if not RaidTimer then
        return false
    end

    pcall(function()
        Visible = RaidTimer.Visible

local function IsRaidTimerSignalVisible()
    -- Fallback para quando o caminho Main/TopHUDList/RaidTimer não é
    -- encontrado em alguma raça/versão da interface.
    -- O contador "Time Left" / "Tempo restante" confirma que a Trial foi aberta.
    if IsRaidTimerVisible() then
        return true
    end

    for _, Text in ipairs(GetTrialResultTexts()) do
        local Normalized = NormalizeTrialText(Text)

        if string.find(Normalized, "time left", 1, true)
            or string.find(Normalized, "tempo restante", 1, true)
        then
            return true
        end
    end

    return false
end

--========================================================
-- PROGRESSO REAL DA V4
--========================================================
-- UpgradeRace/Check retorna 0/5/nil para READY FOR TRIAL.
-- Esse "5" é código de estado e não pode virar 5/10.
--
-- Quando o servidor expõe o progresso em TempleClock/Check,
-- usamos somente campos explicitamente numéricos e com nomes de
-- progresso. Se não houver tal campo, mantemos o progresso anterior
-- em vez de inventar 0/10 ou 5/10.

local function ReadRealV4Progress()
    if not CommF then
        return nil
    end

    local Success, Data = pcall(function()
        return CommF:InvokeServer("TempleClock", "Check")
    end)

    if not Success or typeof(Data) ~= "table" then
        return nil
    end

    local CandidateKeys = {
        "Progress",
        "Progression",
        "V4Progress",
        "TrialProgress",
        "Trials",
        "TrialCount",
        "TrialsCompleted",
        "CompletedTrials",
        "Wins",
        "WinCount"
    }

    for _, Key in ipairs(CandidateKeys) do
        local Value = Data[Key]

        if typeof(Value) == "number"
            and Value >= 0
            and Value <= 10
        then
            print(
                "[NUVEM V4] Progresso real encontrado:",
                tostring(Key),
                "=",
                tostring(Value)
            )

            return math.floor(Value)
        end
    end

    -- Diagnóstico: não assume que qualquer número encontrado seja progresso.
    print("[NUVEM V4] TempleClock/Check não expôs campo de progresso conhecido.")

    return nil
end

local function RefreshRealV4Progress()
    local Progress = ReadRealV4Progress()

    if typeof(Progress) == "number" then
        SetStableV4Progress(Progress, "TEMPLECLOCK")
        return true
    end

    return false
end

local function BuyGearAfterTrial()

    if GearPurchaseBusy then
        return false
    end

    if GearPurchaseDoneForTrial then
        return true
    end

    if not CommF then
        warn("[Nuvem V4] CommF_ indisponível para comprar Gear.")
        return false
    end

    local ProgressAtStart = TrialStartProgress

    -- Na primeira Trial do ciclo, o estágio correto é 0/10.
    -- B=5 do Ancient One significa "Ready For Trial", não 5/10.
    if typeof(ProgressAtStart) ~= "number" then
        if typeof(StableV4Progress) == "number" and StableV4Progress >= 0 then
            ProgressAtStart = StableV4Progress
        else
            print("[Nuvem V4] Não há estágio de Gear definido para esta Trial.")
            GearPurchaseDoneForTrial = true
            return true
        end
    end

    if ProgressAtStart == 5 and StableV4Progress == 0 then
        ProgressAtStart = 0
        print("[Nuvem V4] B=5 era apenas Ready For Trial -> usando estágio 0/10 para Gear1.")
    end

    local Plan = GearPlan[ProgressAtStart]

    -- 5/10: quarta Trial. Não escolhe nenhuma Gear.
    if not Plan then
        print("[Nuvem V4] Estágio", tostring(ProgressAtStart) .. "/10: nenhuma Gear será escolhida.")
        GearPurchaseDoneForTrial = true
        return true
    end

    GearPurchaseBusy = true

    -- O ponto pode aparecer alguns instantes depois do fim/reset da Trial.
    -- Não desistimos no primeiro Check.
    for Attempt = 1, 20 do

        if not Config.Enabled then
            GearPurchaseBusy = false
            return false
        end

        local Success, CheckResult = pcall(function()
            return CommF:InvokeServer("TempleClock", "Check")
        end)

        if Success and typeof(CheckResult) == "table" then

            print(

                .. tostring(Attempt)
                .. " | HadPoint="
                .. tostring(CheckResult.HadPoint)
            )

            if CheckResult.HadPoint then

                print("==============================================")
                print("[Nuvem V4] GEAR DISPONÍVEL APÓS A TRIAL")
                print("[Nuvem V4] Estágio:", tostring(ProgressAtStart) .. "/10")
                print("[Nuvem V4] Comprando:", Plan.Gear)
                print("[Nuvem V4] Cor:", Plan.Color)
                print("[Nuvem V4] Tipo:", Plan.Type)
                print("==============================================")

                local BuySuccess, BuyResult = pcall(function()
                    return CommF:InvokeServer(
                        "TempleClock",
                        "SpendPoint",
                        Plan.Gear,
                        Plan.Type
                    )
                end)

                if BuySuccess then
                    print("[Nuvem V4] Compra enviada:", Plan.Gear, Plan.Type)
                    print("[Nuvem V4] Resultado:", tostring(BuyResult))

                    GearPurchaseDoneForTrial = true
                    GearPurchaseBusy = false

                    UpdateStatusUI(
                        "GEAR COMPRADA",
                        Plan.Gear .. " " .. Plan.Color .. " concluída — consultando Ancient One...",
                        tostring(ProgressAtStart) .. "/10"
                    )

                    return true
                end

                warn(
                    "[Nuvem V4] Falha ao enviar compra de "
                    .. Plan.Gear
                    .. " — tentando novamente."
                )

            else
                print(
                    "[Nuvem V4] Ponto da Gear ainda não liberado. "
                    .. "Aguardando antes de tentar novamente..."
                )
            end

        else
            warn(
                "[Nuvem V4] TempleClock / Check falhou na tentativa "
                .. tostring(Attempt)
                .. " — tentando novamente."
            )
        end

        task.wait(0.5)
    end

    GearPurchaseBusy = false
    V4State = "GEAR_WAITING"

    UpdateStatusUI(
        "AGUARDANDO GEAR",
        "Ponto da Gear ainda não apareceu — tentando novamente...",
        tostring(ProgressAtStart) .. "/10"
    )

    return false

end

local HumanLastM1 = 0

local function ActivateTrialM1()
    local CurrentCharacter = Player.Character
    if not CurrentCharacter then
        return false
    end

    local Humanoid =
        CurrentCharacter:FindFirstChildOfClass("Humanoid")

    if not Humanoid or Humanoid.Health <= 0 then
        return false
    end

    -- Usa exatamente o mesmo método do Auto Training/Bones que já funciona:
    -- equipar o estilo de luta e chamar Tool:Activate() diretamente.
    EquipFightingStyle()
    task.wait()

    local Tool =
        CurrentCharacter:FindFirstChildOfClass("Tool")

    if not Tool or Tool.ToolTip ~= "Melee" then
        return false
    end

    if tick() - HumanLastM1 < 0.08 then
        return true
    end

    HumanLastM1 = tick()

    pcall(function()
        Tool:Activate()
    end)

    return true
end

end

local function TrialAttackSkills()
    -- A Trial Humana é EXCLUSIVAMENTE M1.
    if TrialRace == "Human" then
        return
    end

    if tick() - TrialLastAttack < 0.35 then
        return
    end

    TrialLastAttack = tick()

    EquipFightingStyle()
    ActivateTrialM1()

    PressTrialSkill(Enum.KeyCode.Z)
    PressTrialSkill(Enum.KeyCode.X)
    PressTrialSkill(Enum.KeyCode.C)
end

local function FindTrialEnemy(Names, MaxDistance)
    local Root = GetRoot()

    if not Root then
        return nil
    end

    local Best = nil
    local BestDistance = math.huge

    for _, NPC in ipairs(Enemies:GetChildren()) do

        local ValidName = false

        for _, Name in ipairs(Names) do
            if NPC.Name == Name then
                ValidName = true
                break
            end
        end

        if ValidName and IsAlive(NPC) then

            local NPCRoot =
                NPC:FindFirstChild("HumanoidRootPart")

            if NPCRoot then

                local Distance =
                    (Root.Position - NPCRoot.Position).Magnitude

                if Distance <= MaxDistance
                    and Distance < BestDistance
                then
                    Best = NPC
                    BestDistance = Distance
                end
            end
        end
    end

    return Best
end

--========================================================
-- HUMAN TRIAL - MESMO PADRÃO DO FARM DE BONES
--========================================================
-- IMPORTANTE:
-- Não vamos mais mover o boss para baixo do jogador a cada chamada.
-- Isso fazia a posição do alvo mudar e podia acabar "puxando" o NPC
-- para cima sem que o M1 registrasse dano.
--
-- A lógica agora é:
-- 1) encontra o boss da Trial;
-- 2) salva UMA posição fixa do boss;
-- 3) mantém o boss nessa posição, como o PosMon dos Bones;
-- 4) mantém o jogador fixo 15 studs acima;
-- 5) equipa o melee e chama Tool:Activate() continuamente.

local HumanHitboxPrepared = nil
local HumanTrialLocked = false
local HumanTrialLockedCFrame = nil
local HumanOriginalBossCFrame = nil

local function PrepareHumanTrialHitbox(Boss)
    local BossRoot = Boss and Boss:FindFirstChild("HumanoidRootPart")
    local BossHumanoid = Boss and Boss:FindFirstChildOfClass("Humanoid")

    if not BossRoot or not BossHumanoid or BossHumanoid.Health <= 0 then
        return false
    end

    -- A posição do boss é travada UMA vez, exatamente como PosMon no Bones.
    if HumanHitboxPrepared ~= Boss then
        HumanTrialLocked = false
        HumanTrialLockedCFrame = nil
        HumanOriginalBossCFrame = nil
    end

    if not HumanTrialLocked or not HumanTrialLockedCFrame then
        HumanOriginalBossCFrame = BossRoot.CFrame
        HumanTrialLockedCFrame = BossRoot.CFrame
        HumanTrialLocked = true

        print("==============================================")
        print("[NUVEM V4 HUMAN] BOSS TRAVADO NO POSMON")
        print("[NUVEM V4 HUMAN] Boss:", tostring(Boss.Name))
        print("[NUVEM V4 HUMAN] Posição:", tostring(HumanTrialLockedCFrame.Position))
        print("==============================================")
    end

    HumanHitboxPrepared = Boss

    pcall(function()
        BossRoot.CanCollide = false
        BossRoot.Anchored = false
        BossRoot.AssemblyLinearVelocity = Vector3.zero
        BossRoot.AssemblyAngularVelocity = Vector3.zero
        -- Mesmo princípio do Magnet/BringBone: root maior para o M1.
        BossRoot.Size = Vector3.new(8, 8, 8)
    end)

    return true
end

ReleaseHumanTrialHitbox = function()
    local Boss = HumanHitboxPrepared
    HumanHitboxPrepared = nil

    if Boss and Boss.Parent then
        local BossRoot = Boss:FindFirstChild("HumanoidRootPart")
        if BossRoot then
            pcall(function()
                BossRoot.Anchored = false
                BossRoot.CanCollide = true
                BossRoot.Size = Vector3.new(2, 2, 1)
                BossRoot.AssemblyLinearVelocity = Vector3.zero
                BossRoot.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    HumanTrialLocked = false
    HumanTrialLockedCFrame = nil
    HumanOriginalBossCFrame = nil
end

local function MaintainHumanTrialPosition()
    if not HumanHitboxPrepared
        or not HumanHitboxPrepared.Parent
        or not HumanTrialLocked
        or not HumanTrialLockedCFrame
    then
        return false
    end

    local Boss = HumanHitboxPrepared
    local BossHumanoid = Boss:FindFirstChildOfClass("Humanoid")
    local BossRoot = Boss:FindFirstChild("HumanoidRootPart")
    local Root = GetRoot()

    if not BossHumanoid or BossHumanoid.Health <= 0 then
        return false
    end

    if not BossRoot or not Root then
        return false
    end

    -- EXATAMENTE como o BringBoneMob: NPC no PosMon salvo.
    pcall(function()
        BossHumanoid.WalkSpeed = 0
        BossHumanoid.JumpPower = 0

        BossRoot.CanCollide = false
        BossRoot.CFrame = HumanTrialLockedCFrame
        BossRoot.AssemblyLinearVelocity = Vector3.zero
        BossRoot.AssemblyAngularVelocity = Vector3.zero

        -- Jogador FIXO acima do boss.
        local PlayerPosition =
            HumanTrialLockedCFrame.Position
            + Vector3.new(0, Config.FarmHeight, 0)

        Root.CFrame = CFrame.lookAt(
            PlayerPosition,
            HumanTrialLockedCFrame.Position
        )

        Root.AssemblyLinearVelocity = Vector3.zero
        Root.AssemblyAngularVelocity = Vector3.zero
    end)

    return true
end

local function RunHumanTrial()
    local BossNames = {
        "Flame Master",
        "Quake Master",
        "Light Master",
        "Darkness Master",
        "Rumble Master"
    }

    local Boss = HumanHitboxPrepared

    if not Boss
        or not Boss.Parent
        or not IsAlive(Boss)
    then
        Boss = FindTrialEnemy(BossNames, 1500)
    end

    if not Boss then
        print("[NUVEM V4 HUMAN] Boss da Trial ainda não encontrado.")
        return
    end

    if not PrepareHumanTrialHitbox(Boss) then
        return
    end

    -- Mantém boss + jogador na posição fixa antes de cada M1.
    if not MaintainHumanTrialPosition() then
        return
    end

    EquipFightingStyle()

    -- Mesmo ataque usado pelo Auto Training/Bones.
    local Hit = ActivateTrialM1()

    if Hit then
        local BossHumanoid = Boss:FindFirstChildOfClass("Humanoid")
        print(
            "[NUVEM V4 HUMAN] M1 enviado | boss=",
            tostring(Boss.Name),
            "| vida=",
            BossHumanoid and tostring(math.floor(BossHumanoid.Health)) or "nil"
        )
    else
        warn("[NUVEM V4 HUMAN] Não foi possível ativar o M1.")
    end
end

local function RunGhoulTrial()
    local Enemy = FindTrialEnemy(
        {
            "Ancient Vampire",
            "Ancient Zombie"
        },
        1000
    )

    if not Enemy then
        return
    end

    local EnemyRoot =
        Enemy:FindFirstChild("HumanoidRootPart")

    if not EnemyRoot then
        return
    end

    local Target =
        EnemyRoot.CFrame
        * CFrame.new(0, 30, 0)

    MoveTo(Target)

    if
        (GetRoot().Position - Target.Position).Magnitude
        <= 35
    then
        TrialAttackSkills()
    end
end

local function RunSharkTrial()
    local TrialLocation =
        Workspace:FindFirstChild("_WorldOrigin")
        and Workspace._WorldOrigin:FindFirstChild("Locations")
        and Workspace._WorldOrigin.Locations:FindFirstChild(
            "Trial of Water"
        )

    if not TrialLocation then
        return
    end

    local SeaBeasts =
        Workspace:FindFirstChild("SeaBeasts")

    if not SeaBeasts then
        return
    end

    local Best = nil
    local BestDistance = math.huge

    for _, SeaBeast in ipairs(SeaBeasts:GetChildren()) do

        local RootPart =
            SeaBeast:FindFirstChild("HumanoidRootPart")

        local Health =
            SeaBeast:FindFirstChild("Health")

        if RootPart
            and Health
            and Health.Value > 0
        then

            local Distance =
                (RootPart.Position - TrialLocation.Position).Magnitude

            if Distance <= 1500
                and Distance < BestDistance
            then
                Best = SeaBeast
                BestDistance = Distance
            end
        end
    end

    if not Best then
        return
    end

    local BeastRoot =
        Best:FindFirstChild("HumanoidRootPart")

    if not BeastRoot then
        return
    end

    local WaterPlane =
        Workspace.Map:FindFirstChild("WaterBase-Plane")

    if not WaterPlane then
        return
    end

    -- Mesmo princípio do Iorgute:
    -- primeiro vai ao ponto de combate acima da água,
    -- depois começa o ataque.
    local TargetPosition =
        Vector3.new(
            BeastRoot.Position.X,
            WaterPlane.Position.Y + 300,
            BeastRoot.Position.Z
        )

    local Target =
        CFrame.new(TargetPosition)

    MoveTo(Target)

    local Root = GetRoot()

    if Root
        and (Root.Position - Target.Position).Magnitude <= 40
    then
        TrialAttackSkills()
    end
end

local function RunCurrentTrial()
    local Race = GetCurrentRace()

    if not Race then
        return
    end

    if Race == "Mink"
        or Race == "Rabbit"
    then
        RunMinkTrial()

    elseif Race == "Fishman"
        or Race == "Shark"
    then
        RunSharkTrial()

    elseif Race == "Skypiea"
        or Race == "Angel"
    then
        RunAngelTrial()

    elseif Race == "Human" then

            "TRIAL",

        RunGhoulTrial()

    elseif Race == "Cyborg" then
        RunCyborgTrial()
    end
end


--========================================================
-- PORTA DA TRIAL
--========================================================
-- IMPORTANTE:
-- Não existe mais um sistema de movimento separado para a porta.
-- A porta usa MoveTo(), o mesmo método que já funciona em Bones.
-- Isso mantém a mesma velocidade, noclip e controle do movimento.
local TrialDoorHoldLocked = false
local TrialRouteState = "IDLE"
local LastDoorArrival = 0
local TrialDoorAnchored = false

SetTrialDoorAnchor = function(enabled)
    local Root = GetRoot()

    if not Root then
        return
    end

    pcall(function()
        Root.Anchored = enabled
    end)

    TrialDoorAnchored = enabled
end

local function GoToTrialDoor()

    if not TempleReady then
        return false
    end

    -- Depois que chegou, não reinicia o tween nem tenta reposicionar
    -- o personagem a cada ciclo.
    if TrialDoorReady
        or TrialDoorHoldLocked
        or TrialRouteState == "WAITING_AT_DOOR"
    then
        return true
    end

    local TargetCFrame = GetTrialDoorCFrame()

    if not TargetCFrame then
        TrialDoorReady = false

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            "Não foi possível identificar a raça",
            GetProgressText()
        )

        warn(
            "[Nuvem V4] Não encontrei a posição da porta para:",
            tostring(Player.Data.Race.Value)
        )

        return false
    end

    -- O teste isolado que funcionou usou a porta com +6.482 no eixo Y.
    -- Mantemos X/Z e rotação da porta, alterando somente a altura.
    TargetCFrame = TargetCFrame + Vector3.new(0, Config.TrialDoorYOffset, 0)

    local Root = GetRoot()

    if not Root then
        return false
    end

    local RaceName = tostring(Player.Data.Race.Value)

    if Config.DebugTrialRoute then
        print("==============================================")
        print("[NUVEM V4 DIAG] TEMPLE -> PORTA | INÍCIO/ATUALIZAÇÃO")
        print("[NUVEM V4 DIAG] Race:", RaceName)
        print("[NUVEM V4 DIAG] Root:", tostring(Root.Position))
        print("[NUVEM V4 DIAG] Target:", tostring(TargetCFrame.Position))
        print("[NUVEM V4 DIAG] Distance:", (Root.Position - TargetCFrame.Position).Magnitude)
        print("[NUVEM V4 DIAG] TravelSpeed:", Config.TravelSpeed)
        print("[NUVEM V4 DIAG] TrialDoorSpeed (não usado):", Config.TrialDoorSpeed)
        print("[NUVEM V4 DIAG] TempleReady:", tostring(TempleReady))
        print("[NUVEM V4 DIAG] TrialDoorReady:", tostring(TrialDoorReady))
        print("==============================================")
    end

    local Distance =
        (Root.Position - TargetCFrame.Position).Magnitude

    if Distance <= Config.TargetDistance then

        StopMovement()
        SetTrialDoorAnchor(false)

        TrialMoving = false
        TrialDoorReady = true
        TrialDoorHoldLocked = true
        TrialRouteState = "WAITING_AT_DOOR"
        LastDoorArrival = tick()

        -- Um único ajuste final.
        Root.CFrame = TargetCFrame
        Root.AssemblyLinearVelocity = Vector3.zero
        Root.AssemblyAngularVelocity = Vector3.zero

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            "Na porta da Trial — aguardando",
            GetProgressText()
        )

        print(
            "[Nuvem V4] Porta alcançada:",
            RaceName,
            "| ROTA TRAVADA: aguardando Trial."
        )

        if Config.DebugTrialRoute then
            print("[NUVEM V4 DIAG] >>> CHEGOU NA PORTA <<<")
            print("[NUVEM V4 DIAG] Pos final:", tostring(Root.Position))
            print("[NUVEM V4 DIAG] Target final:", tostring(TargetCFrame.Position))
            print("[NUVEM V4 DIAG] Distância final:", tostring((Root.Position - TargetCFrame.Position).Magnitude))
            print("==============================================")
        end

        return true
    end

    TrialDoorReady = false
    TrialMoving = true
    TrialRouteState = "GOING_TO_DOOR"
    TrialDoorHoldLocked = false
    ShouldTween = true
    TrialDiagLast = Root.Position
    TrialDiagLastTime = tick()
    TrialDiagNextPrint = 0

    if Config.DebugTrialRoute then
        print("[NUVEM V4 DIAG] >>> MOVIMENTO TEMPLE -> PORTA INICIADO <<<")
        print("[NUVEM V4 DIAG] Origem:", tostring(Root.Position))
        print("[NUVEM V4 DIAG] Destino:", tostring(TargetCFrame.Position))
        print("[NUVEM V4 DIAG] Distância:", tostring(Distance))
        print("[NUVEM V4 DIAG] Velocidade configurada:", tostring(Config.TravelSpeed))
    end

    -- O teste isolado funcionou sem ancorar o HumanoidRootPart.
    -- Deixamos a física normal e o MoveTo()/Heartbeat controlarem o CFrame.
    SetTrialDoorAnchor(false)

    print(
        "[Nuvem V4] Porta TESTE | Y ajustado:",
        tostring(TargetCFrame.Position.Y),
        "| Offset Y:",
        tostring(Config.TrialDoorYOffset)
    )

    UpdateStatusUI(
        "PRONTO PARA TRIAL",
        "Indo para a porta da Trial (" .. RaceName .. ")",
        GetProgressText()
    )

    print(
        "[Nuvem V4] Indo para porta uma única vez.",
        "Raça:", RaceName,
        "| Distância:", math.floor(Distance),
        "| Velocidade:", Config.TrialDoorSpeed
    )

    --============================================================
    -- MOVIMENTO DIRETO TEMPLE -> PORTA
    --============================================================
    -- Não usamos MoveTo()/CurrentTween aqui.
    -- O controle direto por Heartbeat evita que outra rotina do
    -- controlador principal reinicie/cancele o tween e deixe a
    -- velocidade real muito abaixo da configurada.
    if TrialDoorConnection then
        pcall(function()
            TrialDoorConnection:Disconnect()
        end)
        TrialDoorConnection = nil
    end

    local DoorSpeed = tonumber(Config.TrialDoorSpeed) or 230
    local LastTime = tick()

    TrialDoorConnection = RunService.Heartbeat:Connect(function()
        if not Config.Enabled or not TrialMoving then
            if TrialDoorConnection then
                pcall(function()
                    TrialDoorConnection:Disconnect()
                end)
                TrialDoorConnection = nil
            end
            return
        end

        local CurrentRoot = GetRoot()
        if not CurrentRoot then
            return
        end

        local Now = tick()
        local DeltaTime = math.clamp(Now - LastTime, 0, 0.1)
        LastTime = Now

        local CurrentPosition = CurrentRoot.Position
        local MoveTarget = TargetCFrame

        -- A Trial da Mink é mais sensível à física durante a rota.
        -- Em vez de seguir uma diagonal 3D desde o Temple até a porta,
        -- mantemos o personagem em uma altura segura e só fazemos a
        -- descida quando já estamos próximos da porta.
        if RaceName == "Mink" or RaceName == "Rabbit" then
            local HorizontalOffset = Vector3.new(
                TargetCFrame.Position.X - CurrentPosition.X,
                0,
                TargetCFrame.Position.Z - CurrentPosition.Z
            )

            local HorizontalDistance = HorizontalOffset.Magnitude

            if HorizontalDistance > Config.MinkFinalApproachDistance then
                local SafeY = math.max(
                    CurrentPosition.Y,
                    TargetCFrame.Position.Y + Config.MinkSafeHeightOffset
                )

                MoveTarget = CFrame.new(
                    TargetCFrame.Position.X,
                    SafeY,
                    TargetCFrame.Position.Z
                )
            end
        end

        local Offset = MoveTarget.Position - CurrentPosition
        local Remaining = Offset.Magnitude

        if Remaining <= Config.TargetDistance then
            TrialMoving = false
            TrialDoorReady = true
            TrialDoorHoldLocked = true
            TrialRouteState = "WAITING_AT_DOOR"
            LastDoorArrival = tick()

            CurrentRoot.CFrame = TargetCFrame
            CurrentRoot.AssemblyLinearVelocity = Vector3.zero
            CurrentRoot.AssemblyAngularVelocity = Vector3.zero

            if TrialDoorConnection then
                pcall(function()
                    TrialDoorConnection:Disconnect()
                end)
                TrialDoorConnection = nil
            end

            UpdateStatusUI(
                "PRONTO PARA TRIAL",
                "Na porta da Trial — aguardando",
                GetProgressText()
            )

            print(
                "[Nuvem V4] Porta alcançada:",
                RaceName,
                "| Velocidade usada:", DoorSpeed
            )

            return
        end

        local Step = math.min(DoorSpeed * DeltaTime, Remaining)
        local Direction = Offset.Unit
        local NewPosition = CurrentRoot.Position + Direction * Step

        CurrentRoot.CFrame = CFrame.lookAt(
            NewPosition,
            NewPosition + Direction
        )

        CurrentRoot.AssemblyLinearVelocity = Vector3.zero
        CurrentRoot.AssemblyAngularVelocity = Vector3.zero
    end)

    print(
        "[Nuvem V4] Movimento DIRETO Temple -> porta iniciado.",
        "| Velocidade:", DoorSpeed,
        "| Distância:", math.floor(Distance)
    )

    return false
end

--========================================================
-- V3 - HABILIDADE DA RAÇA PARA INICIAR A TRIAL
--========================================================
-- IMPORTANTE:
-- T = habilidade V3 da raça.
-- Y = ativação V4, usada separadamente no treinamento.
--
-- A V3 só é acionada quando a conta está na preparação da Trial
-- e já chegou à porta/recebeu o sinal do ciclo. Não é chamada durante
-- o treinamento V4.

ActivateV3ForTrial = function(ForceStart)
    if not Config.AutoV3 then
        return false
    end

    if V3ActivatedForTrial then
        return true
    end

    if not ForceStart and V4State ~= "TRIAL" then
        return false
    end

    if TrialRunning then
        return false
    end

    local CharacterNow = Player.Character
    if not CharacterNow then
        return false
    end

    -- A V3 é a habilidade da raça (T).
    -- Não usamos CommE aqui: o método que já funcionou nas versões
    -- anteriores é a entrada real da tecla T.
    if tick() - LastV3Activation < 0.5 then
        return false
    end

    LastV3Activation = tick()

    print("[NUVEM V4 V3] Tentando ativar V3 | raça=" .. tostring(GetCurrentRace()))

    local Success = pcall(function()
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.T, false, game)
        task.wait(0.06)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.T, false, game)
    end)

    if not Success then
        warn("[NUVEM V4 V3] Falha ao enviar tecla T.")
        return false
    end

    -- Não marcamos como concluída antes de dar tempo para o jogo processar.
    -- O latch é local por ciclo; uma nova Trial libera novamente.
    V3ActivatedForTrial = true

    print("==============================================")
    print("[NUVEM V4] HABILIDADE V3 SOLICITADA")
    print("[NUVEM V4] Método: tecla T")
    print("[NUVEM V4] Raça:", tostring(GetCurrentRace()))
    print("==============================================")

    return true
end

local function UpgradeRace()

    if not Config.AutoUpgrade then
        return
    end

    if not CommF then
        return
    end

    if V4State ~= "BUY_AVAILABLE" then
        return
    end

    if tick() - LastUpgrade <
        Config.UpgradeCooldown
    then
        return
    end

    LastUpgrade = tick()

    local Success, Result =
        pcall(function()

            return CommF:InvokeServer(
                "UpgradeRace",
                "Buy"
            )

        end)

    if Success and Result then

        print(
            "[NUVEM V4] Pagamento enviado com sucesso."
        )

        -- O pagamento altera o estado do Ancient One.
        -- Força uma nova consulta logo depois para não ficar
        -- preso no estado anterior/UNKNOWN.
        AncientOneMessage = "Pagamento concluído — consultando Ancient One..."
        print("[NUVEM V4] Pagamento concluído. Consultando Ancient One novamente...")

        LastV4Check = 0

        task.delay(0.4, function()
            if Config.Enabled then
                LastV4Check = 0
            end
        end)

    elseif not Success then

        warn(
            "[NUVEM V4] Erro ao pagar:",
            Result
        )

    end

end

--========================================================
-- CONSULTA DO ANCIENT ONE
--========================================================
--
-- O retorno de UpgradeRace / Check é o estado que o jogo usa
-- para representar o que o Ancient One está indicando.
-- Mantemos a resposta completa A/B/C para não perder informação.
--========================================================

local function InterpretAncientOne(A, B, C)

    AncientOneCode = A

    -- CONFIRMADO NO TESTE:
    -- A = nil / B = nil / C = nil
    -- Este retorno é V3 no diagnóstico atual.
    if A == nil and B == nil and C == nil then
        return "V3"
    end

    -- CONFIRMADO NO TESTE:
    -- A = 0 / B = 5 / C = nil
    -- Ancient One: Ready For Trial
    if A == 0 and B == 5 and C == nil then
        return "READY FOR TRIAL"
    end

    -- Mantém compatibilidade caso o jogo retorne outro contador
    -- para o mesmo estado.
    if A == 0 and (B == 1 or B == 2) then
        return "READY FOR TRIAL"
    end

    -- Raça completa.
    if A == 5 and B == 10 then
        return "RAÇA COMPLETA"
    end

    -- CONFIRMADO NOS TESTES:
    -- A = 6 / B = 2 / C = nil
    -- Ancient One: precisa treinar.
    if A == 6 and typeof(B) == "number" and C == nil then
        return "PRECISA TREINAR"
    end

    -- Outros estados de treinamento sem preço.
    if typeof(A) == "number"
        and typeof(B) == "number"
        and A > 0
        and B > 0
        and B < 10
        and C == nil
    then
        return "PRECISA TREINAR"
    end

    -- Compra disponível quando o servidor devolve um preço.
    if typeof(C) == "number" and C > 0 then
        return "COMPRA DISPONÍVEL"
    end

    return "FALA/ESTADO NÃO MAPEADO"
end

local function CheckV4Status()

    if not CommF then

        V4State = "UNKNOWN"

        UpdateStatusUI(
            "UNKNOWN",
            "CommF_ não encontrado",
            "--"
        )

        return V4State
    end

    -- Durante a Trial, o retorno do Ancient One não deve
    -- sobrescrever o estado TRIAL_RUNNING.
    if V4State == "WAITING_TRIAL_TELEPORT"
        or V4State == "TRIAL_RUNNING"
        or V4State == "GEAR_WAITING"
    then
        return V4State
    end

    if V4CheckBusy then
        return V4State
    end

    if tick() - LastV4Check < 2 then
        return V4State
    end

    V4CheckBusy = true
    LastV4Check = tick()

    print("")
    print("==============================================")
    print("[NUVEM V4] DIAGNÓSTICO INICIADO")
    print("==============================================")
    print("[NUVEM V4] CONSULTANDO ANCIENT ONE")
    print("[NUVEM V4] UpgradeRace / Check...")

    local Success, A, B, C =
        pcall(function()

            return CommF:InvokeServer(
                "UpgradeRace",
                "Check"
            )

        end)

    if not Success then

        V4State = "UNKNOWN"
        -- Se o Ancient One mudou para um estado real, liberamos
        -- um novo ciclo de Trial.
        SuppressReadyForTrial = false

        V4Progress = StableV4Progress
        V4SessionsLeft = 0
        V4Price = nil
        V4CanFarmBones = false

        StopMovement()
        StopTrialDoorMovement()
        CurrentTarget = nil

        warn(
            "[NUVEM V4] Erro ao consultar Ancient One:",
            A
        )

        UpdateStatusUI(
            "UNKNOWN",
            "Erro ao consultar Ancient One",
            "--"
        )

        V4CheckBusy = false

        return V4State
    end

    print("")
    print("[NUVEM V4] RESPOSTA DO ANCIENT ONE")
    print("----------------------------------------------")

    print(
        "[V4] Retorno 1:",
        tostring(A),
        "| Tipo:",
        typeof(A)
    )

    print(
        "[V4] Retorno 2:",
        tostring(B),
        "| Tipo:",
        typeof(B)
    )

    print(
        "[V4] Retorno 3:",
        tostring(C),
        "| Tipo:",
        typeof(C)
    )

    print("----------------------------------------------")

    -- A=6/B=n é usado pelo próprio UpgradeRace como estado de treinamento.
    -- A=0/B=5/nil é apenas READY FOR TRIAL e NÃO é progresso.
    if A == 6 and type(B) == "number" and C == nil then
        SetStableV4Progress(B, "UPGRADERACE_CHECK")

        print(
            "[V4] Progresso real confirmado pelo UpgradeRace:",
            tostring(B) .. "/10"
        )
    elseif type(C) == "number" and C > 0 and type(B) == "number" then
        SetStableV4Progress(B, "UPGRADERACE_BUY_AVAILABLE")

        print(
            "[V4] Progresso real confirmado antes da compra:",
            tostring(B) .. "/10"
        )
    end

    AncientOneMessage = InterpretAncientOne(A, B, C)

    print("[V4] Ancient One:", AncientOneMessage)
    print(
        "[V4] Dados completos:",
        tostring(A),
        "/",
        tostring(B),
        "/",
        tostring(C)
    )

    --====================================================
    -- V3 / NIL NIL NIL -> PRIMEIRA TRIAL
    --====================================================
    -- A versão funcional já confirmou que nil/nil/nil é V3.
    -- NÃO pode cair no bloco UNKNOWN.
    if A == nil and B == nil and C == nil then

        -- IMPORTANTE:
        -- nil/nil/nil é V3, mas o diagnóstico roda em loop.
        -- Portanto, NÃO podemos resetar TempleReady/TrialDoorReady
        -- em toda consulta, senão a conta volta para "indo ao Temple"
        -- infinitamente e nunca estabiliza na porta.
        local WasAlreadyV3 =
            V4CycleStarted
            and (
                V4State == "TRIAL"
                or V4State == "WAITING_TRIAL_TELEPORT"
                or V4State == "TRIAL_RUNNING"
                or TrialDoorReady
                or TempleReady
            )

        if not V4CycleStarted then
            V4CycleStarted = true
            V3PendingTrial = true
            SetStableV4Progress(0, "V3_NIL_NIL_NIL")
            AncientOneProbeRequired = true
            AncientOneProbeDone = false
        end

        V4State = "TRIAL"
        V4SessionsLeft = 0
        V4Price = nil
        V4CanFarmBones = false

        if not WasAlreadyV3 then
            TempleReady = false
            TempleEntering = false
            TrialDoorReady = false

            StopMovement()
            StopTrialDoorMovement()
            CurrentTarget = nil

            print("[V4] ESTADO = RAÇA V3")
            print("[V4] Check = nil / nil / nil")
            print("[V4] Primeira confirmação V3 — preparando Temple.")
        else
            print("[V4] V3 confirmada novamente — mantendo rota atual.")
            print("[V4] TempleReady:", tostring(TempleReady),
                  "| TrialDoorReady:", tostring(TrialDoorReady))
        end

        local V3Action

        if not TempleReady then
            V3Action = "Indo para o Temple of Time..."
        elseif not TrialDoorReady then
            V3Action = "Temple alcançado — indo para a porta da Trial..."
        else
            V3Action = "Na porta da Trial — aguardando"
        end

        print("[V4] Progresso estável:", GetProgressText())

        UpdateStatusUI(
            "RAÇA V3",
            V3Action,
            GetProgressText()
        )

        NotifyAncientOne(
            "TRIAL",
            "Raça V3 detectada — " .. V3Action
        )

    --====================================================
    -- V3 / 0-10 -> PRIMEIRA TRIAL (compatibilidade)
    --====================================================
    elseif A == 0 and B == 0 and C == nil then

        -- Esse retorno representa a conta que está no início do ciclo V4.
        -- Só inicializamos 0/10 se ainda não começamos o ciclo.
        -- Depois de uma Trial, um retorno transitório 0/0 não pode
        -- apagar o progresso que já foi conquistado.
        if not V4CycleStarted then
            V4CycleStarted = true
            SetStableV4Progress(0, "V3_INITIAL")
            AncientOneProbeRequired = true
            AncientOneProbeDone = false
        else
            print("[V4] Retorno 0/0 recebido novamente; mantendo progresso estável:", GetProgressText())
        end

        V4State = "TRIAL"
        V4SessionsLeft = 0
        V4Price = nil
        V4CanFarmBones = false

        TempleReady = false
        TempleEntering = false
        TrialDoorReady = false

        StopMovement()
        StopTrialDoorMovement()
        CurrentTarget = nil

        print("[V4] ESTADO = V3 / início do ciclo — PRIMEIRA TRIAL")
        print("[V4] Progresso estável:", GetProgressText())

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            "Primeira Trial — preparando Temple of Time...",
            GetProgressText()
        )

        NotifyAncientOne(
            "TRIAL",
            "Início do ciclo V4 — preparando a primeira Trial."
        )

    --====================================================
    -- READY FOR TRIAL
    -- CONFIRMADO: A = 0 / B = 5 / C = nil
    --====================================================

    elseif A == 0 and B == 5 and C == nil then

        -- SINCRONIZAÇÃO:
        -- Depois que uma conta terminou a Trial, o Ancient One pode continuar
        -- retornando READY FOR TRIAL por alguns instantes. Não usamos isso para
        -- mandar a conta novamente à porta. O servidor de sincronização decide
        -- quando o grupo pode se mover.
        if V4Sync.Enabled
            and SyncRole ~= "UNKNOWN"
            and SyncTrialReported
            and (
                SyncState == "TRIAL_DONE"
                or SyncCommand == "WAIT_TRIALS"
                or SyncCommand == "RESET_AUX"
                or SyncCommand == "WAIT_AUX_RESET"
                or SyncPostResetRoute
            )
        then
            V4State = "SYNC_WAIT"
            StopMovement()
            StopTrialDoorMovement()
            TrialDoorReady = false

            UpdateStatusUI(
                "AGUARDANDO GRUPO",
                SyncRole == "MAIN"
                    and "Trial concluída — aguardando as auxiliares..."
                    or "Trial concluída — aguardando o grupo...",
                GetProgressText()
            )

            print("[V4] READY FOR TRIAL ignorado durante sincronização do grupo.")
            V4CheckBusy = false
            return V4State
        end

        -- Depois que uma Trial terminou, aguardamos o Ancient One
        -- realmente mudar de estado antes de permitir outra rota.
        if SuppressReadyForTrial then
            print("[V4] Ready For Trial ainda presente após término; aguardando mudança real do Ancient One.")
            V4CheckBusy = false
            return V4State
        end

        -- O diagnóstico roda continuamente.
        -- Não podemos reiniciar a rota em cada consulta.
        local WasAlreadyTrial = (V4State == "TRIAL")

        V4State = "TRIAL"
        V4CycleStarted = true

        -- B=5 é apenas o código do estado Ready For Trial.
        -- NUNCA usamos B=5 como progresso da Gear.
        V4Progress = StableV4Progress
        V4SessionsLeft = 0
        V4Price = nil
        V4CanFarmBones = false

        -- Só inicializa a rota quando entra no estado TRIAL.
        if not WasAlreadyTrial then

            TempleReady = false
            TempleEntering = false
            TrialDoorReady = false

            StopMovement()
            StopTrialDoorMovement()
            CurrentTarget = nil

            print("[V4] ESTADO = READY FOR TRIAL")
            print("[V4] Ancient One: Ready For Trial")

            -- B=5 não é o progresso da Gear. Descobrimos a fonte real
            -- do progresso V4 olhando o retorno do TempleClock sem assumir
            -- um campo que o servidor não confirmou.
            pcall(function()
                local Ok, Data = pcall(function()
                    return CommF:InvokeServer("TempleClock", "Check")
                end)

                if Ok and typeof(Data) == "table" then
                    print("========== [V4] TEMPLECLOCK CHECK ==========")
                    for Key, Value in pairs(Data) do
                        print(
                            "[TempleClock]",
                            tostring(Key),
                            "=",
                            tostring(Value),
                            "| Tipo:",
                            typeof(Value)
                        )
                    end
                    print("==============================================")
                else
                    print("[V4] TempleClock/Check não retornou tabela.")
                end
            end)

            NotifyAncientOne(
                "TRIAL",
                "Ready For Trial — indo para a porta da Trial."
            )

        else

            print(
                "[V4] READY FOR TRIAL confirmado — mantendo rota atual."
            )

        end

        -- READY FOR TRIAL:
        -- A=0/B=5/nil é somente o código do estado Ready For Trial.
        -- B=5 NÃO representa 5/10.
        --
        -- Se não houver um campo real de progresso no TempleClock, não
        -- inventamos um valor. Mantemos o último progresso conhecido.
        if V3PendingTrial then
            SetStableV4Progress(0, "V3_READY_FOR_TRIAL")
            V3PendingTrial = false
        else
            V4Progress = StableV4Progress

            print(
                "[V4] READY FOR TRIAL -> mantendo progresso real:",
                GetProgressText()
            )
        end

        -- O TempleClock continua sendo consultado apenas como fonte
        -- complementar; não deixa campos desconhecidos sobrescreverem
        -- o estágio já determinado pelo diagnóstico.
        RefreshRealV4Progress()

        local TrialAction

        if not TempleReady then
            TrialAction = "Indo para o Temple of Time..."
        elseif not TrialDoorReady then
            TrialAction = "Temple alcançado — indo para a porta da Trial..."
        else
            TrialAction = "Na porta da Trial — aguardando"
        end

        local ReadyProgress = "--"

        if StableV4Progress > 0 then
            ReadyProgress = GetProgressText()
        end

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            TrialAction,
            ReadyProgress
        )

    -- Compatibilidade com possíveis retornos antigos do mesmo estado.
    elseif A == 0 and (B == 1 or B == 2) and C == nil then

        if SuppressReadyForTrial then
            print("[V4] Ready For Trial ainda presente após término; aguardando mudança real do Ancient One.")
            V4CheckBusy = false
            return V4State
        end

        local WasAlreadyTrial = (V4State == "TRIAL")

        V4State = "TRIAL"
        V4CycleStarted = true

        -- B=5 aqui é código do estado READY FOR TRIAL, não 5 Gears.
        -- Mantém o progresso real já conhecido; se ainda não houver um
        -- progresso real, fica em 0/10 em vez de exibir 5/10.
        if StableV4Progress <= 0 then
            SetStableV4Progress(0, "READY_COMPAT")
        end
        V4SessionsLeft = 0
        V4Price = nil
        V4CanFarmBones = false

        if not WasAlreadyTrial then

            TempleReady = false
            TempleEntering = false
            TrialDoorReady = false

            StopMovement()
            StopTrialDoorMovement()
            CurrentTarget = nil

            print("[V4] ESTADO = READY FOR TRIAL")

            NotifyAncientOne(
                "TRIAL",
                "Ready For Trial — indo para a porta da Trial."
            )

        else

            print(
                "[V4] READY FOR TRIAL confirmado — mantendo rota atual."
            )

        end

        local TrialAction

        if not TempleReady then
            TrialAction = "Indo para o Temple of Time..."
        elseif not TrialDoorReady then
            TrialAction = "Temple alcançado — indo para a porta da Trial..."
        else
            TrialAction = "Na porta da Trial — aguardando"
        end

        local ReadyProgress = "--"

        if StableV4Progress > 0 then
            ReadyProgress = GetProgressText()
        end

        UpdateStatusUI(
            "PRONTO PARA TRIAL",
            TrialAction,
            ReadyProgress
        )

    --====================================================
    -- RAÇA COMPLETA
    -- CONFIRMADO: A = 5 / B = 10 / C = nil
    --====================================================

    elseif A == 5 and B == 10 then

        SetStableV4Progress(10, "RACA_COMPLETA")
        V4State = "COMPLETE"
        SuppressReadyForTrial = false
        V4CycleStarted = true

        SetStableV4Progress(10, "COMPLETE")
        V4SessionsLeft = 0
        V4Price = nil
        V4CanFarmBones = false

        TempleReady = false
        TempleEntering = false
        TrialDoorReady = false

        StopMovement()
        StopTrialDoorMovement()
        CurrentTarget = nil

        print("[V4] ESTADO = RAÇA COMPLETA")

        UpdateStatusUI(
            "RAÇA COMPLETA",
            "Nenhuma ação necessária",
            "10/10"
        )

        NotifyAncientOne(
            "COMPLETE",
            "A raça V4 está completa."
        )

    --====================================================
    -- TREINAMENTO
    -- CONFIRMADO NO TESTE: A = 6 / B = 2 / C = nil
    --====================================================

    elseif A == 6
        and typeof(B) == "number"
        and C == nil
    then

        V4State = "TRAINING"
        SuppressReadyForTrial = false
        V4CycleStarted = true

        SetStableV4Progress(B, "TRAINING")

        V4SessionsLeft =
            math.max(
                1,
                10 - B
            )

        V4Price = nil

-- DETECTOR DE RESULTADO DA TRIAL

-- Algumas versões do Blox Fruits exibem o resultado no CoreGui.
-- Mantemos PlayerGui como fallback e normalizamos o texto para
-- evitar falhas por quebra de linha/espaços.

local function NormalizeTrialText(Text)
    Text = tostring(Text or "")
    Text = string.lower(Text)
    Text = Text:gsub("[%c\r\n\t]+", " ")
    Text = Text:gsub("%s+", " ")
    return Text
end

local function GetTrialResultTexts()
    local Results = {}
    local Seen = {}

    local function ScanRoot(Root)
        if not Root then
            return
        end

        local Objects = {}
        pcall(function()
            Objects = Root:GetDescendants()
        end)

        for _, Object in ipairs(Objects) do
            if Object:IsA("TextLabel")
                or Object:IsA("TextButton")
                or Object:IsA("TextBox")
            then
                local Visible = true
                local Text = ""

                pcall(function()
                    Visible = Object.Visible == true
                    Text = tostring(Object.Text or "")
                end)

                -- Resultado de Trial só deve ser lido enquanto o texto estiver
                -- realmente visível. Isso evita reutilizar mensagens antigas que
                -- ficaram escondidas no CoreGui/PlayerGui.
                if Visible and Text ~= "" then
                    local Normalized = NormalizeTrialText(Text)
                    if not Seen[Normalized] then
                        Seen[Normalized] = true
                        table.insert(Results, Normalized)
                    end
                end
            end
        end
    end

    ScanRoot(game:GetService("CoreGui"))
    ScanRoot(Player:FindFirstChild("PlayerGui"))

    return Results
end

local function IsTrialFailureVisible()
    for _, Text in ipairs(GetTrialResultTexts()) do
        if string.find(Text, "a player failed a trial", 1, true)
            or string.find(Text, "you failed your trial", 1, true)
            or string.find(Text, "you failed a trial", 1, true)
            or string.find(Text, "failed your trial", 1, true)
            or string.find(Text, "failed a trial", 1, true)
            or string.find(Text, "voce falhou", 1, true)
            or string.find(Text, "você falhou", 1, true)
        then
            return true
        end
    end
    return false
end

local TrialResultDebugAt = 0
local AuxFightDetected = false

local function IsFightVisible()
    if not SyncRole or SyncRole ~= "AUX" then
        return false, nil
    end

    for _, Text in ipairs(GetTrialResultTexts()) do
        local Normalized = string.lower(tostring(Text or ""))

        -- Indicador principal: banner que aparece no início da Trial.
        if Normalized == "fight!"
            or Normalized == "fight"
            or string.find(Normalized, "fight!", 1, true)
        then
            return true, "FIGHT!"
        end

        -- Fallback persistente: durante a batalha o jogo mostra "In Combat".
        if Normalized == "in combat"
            or string.find(Normalized, "in combat", 1, true)
        then
            return true, "IN_COMBAT"
        end

        -- Indicador específico da arena da Trial.
        if string.find(Normalized, "don't leave the game", 1, true)
            or string.find(Normalized, "dont leave the game", 1, true)
        then
            return true, "DONT_LEAVE_THE_GAME"
        end
    end

    return false, nil
end

local function IsTrialFightVisible()
    -- IMPORTANTE:
    -- Para o reset da AUX, somente o banner EXATO "FIGHT!" vale.
    -- "In Combat", "Don't leave the game" e outras mensagens NÃO são
    -- consideradas FIGHT, pois podem aparecer durante a preparação/Trial.
    for _, Text in ipairs(GetTrialResultTexts()) do
        local Normalized = NormalizeTrialText(Text)

        if Normalized == "fight!"
            or Normalized == "fight"
            or string.find(Normalized, "fight!", 1, true)
        then
            return true, "FIGHT!"
        end
    end

    return false, nil
end

local function HandleAuxFight(FightReason)
    if SyncRole ~= "AUX" or AuxFightDetected then
        return false
    end

    -- O monitor já confirmou o banner FIGHT!. A partir daqui o reset é local
    -- e NÃO pode depender de uma resposta HTTP para acontecer.
    AuxFightDetected = true
    SyncStartIssued = false
    V3ActivatedForTrial = false
    SyncTrialReported = false
    SyncResetRequested = true
    SyncPostResetRoute = true
    SyncState = "AUX_FIGHT"

    -- FIGHT! inicia o fluxo de reset imediatamente.
    -- O RESET_OK só será enviado no CharacterAdded após o respawn.
    pcall(function()
        ReleaseHumanTrialHitbox()
    end)

    StopMovement()
    StopTrialDoorMovement()

    UpdateStatusUI(
        "RESETANDO AUXILIAR",
        "Combate detectado — enviando AUX_FIGHT e resetando agora...",
        GetProgressText()
    )

    print("[Nuvem V4 SYNC] COMBATE DA TRIAL detectado na AUX -> AUX_FIGHT -> reset local | motivo=" .. tostring(FightReason or "FIGHT!"))

    -- IMPORTANTE: SyncReport pode falhar/lançar erro. Isso nunca pode impedir
    -- o Humanoid.Health = 0. O envio ao servidor é protegido por pcall.
    local ReportOk, Data = pcall(function()
        return SyncReport("AUX_FIGHT", "fight_detected_reset")
    end)

    if ReportOk and Data then
        print("[Nuvem V4 SYNC] AUX_FIGHT enviado | phase=" .. tostring(Data.phase)
            .. " | command=" .. tostring(Data.command)
            .. " | cycle=" .. tostring(Data.cycleId))
    elseif not ReportOk then
        warn("[Nuvem V4 SYNC] Erro ao enviar AUX_FIGHT; reset local continuará normalmente.", Data)
    else
        warn("[Nuvem V4 SYNC] AUX_FIGHT sem confirmação HTTP; reset local continuará normalmente.")
    end

    -- O reset NÃO depende do servidor. FIGHT! já é prova suficiente de que
    -- esta AUX entrou na arena.
    local CharacterNow = Player.Character
    local Humanoid = CharacterNow and CharacterNow:FindFirstChildOfClass("Humanoid")

    if Humanoid then
        print("[Nuvem V4 SYNC] Aplicando Humanoid.Health = 0 após FIGHT! detectado.")
        pcall(function()
            Humanoid.Health = 0
        end)

        -- Fallback apenas se o setter não produzir o estado morto.
        task.delay(0.35, function()
            pcall(function()
                if Humanoid and Humanoid.Parent and Humanoid.Health > 0 and SyncResetRequested then
                    print("[Nuvem V4 SYNC] Health ainda > 0; forçando estado Dead da AUX.")
                    Humanoid:ChangeState(Enum.HumanoidStateType.Dead)
                end
            end)
        end)
    else
        warn("[Nuvem V4 SYNC] Humanoid não encontrado após FIGHT!; aguardando RESET_AUX do servidor.")
    end

    -- Se o POST AUX_FIGHT falhar, tenta novamente sem bloquear o reset.
    task.spawn(function()
        for _ = 1, 8 do
            task.wait(0.75)
            if not Config.Enabled or SyncState == "RESET_OK" then
                return
            end

            local Ok, RetryData = pcall(function()
                return SyncReport("AUX_FIGHT", "fight_detected_reset_retry")
            end)

            if Ok and RetryData then
                print("[Nuvem V4 SYNC] AUX_FIGHT reenviado com sucesso.")
                return
            end
        end
    end)

    return true
end

task.spawn(function()
    while Config.Enabled do
        pcall(function()
            -- FIGHT! é a confirmação de que a batalha realmente começou.
            -- Só aceitamos esse indicador enquanto a conta já estiver no fluxo
            -- da Trial; mensagens antigas fora desse fluxo não iniciam o latch.
            if V4State == "TRIAL_RUNNING" and not TrialBattleStarted then
                local FightStarted, FightReason = IsTrialFightVisible()

                if FightStarted then
                    -- Se FIGHT! já existia quando a Trial foi confirmada,
                    -- não usamos esse texto antigo como gatilho.
                    if TrialFightWasVisibleAtStart then
                        print("[Nuvem V4] FIGHT! antigo ainda visível -> aguardando o banner desaparecer.")
                    else
                        -- O próprio banner FIGHT! é o gatilho da arena.
                        -- NÃO exigimos WAITING_TRIAL_RESULT aqui, porque no
                        -- teste real o servidor ainda pode estar em
                        -- TRIAL_RUNNING quando o FIGHT! aparece.
                        --
                        -- A proteção contra reset no começo da Trial é feita
                        -- pelo snapshot TrialFightWasVisibleAtStart acima:
                        -- se FIGHT! já estava na tela quando a Trial começou,
                        -- ele é ignorado; somente um FIGHT! que surgir depois
                        -- pode disparar o reset da AUX.
                        TrialBattleStarted = true

                        print("[Nuvem V4] >>> FIGHT! FINAL CONFIRMADO <<< | motivo="
                            .. tostring(FightReason)
                            .. " | role=" .. tostring(SyncRole)
                            .. " | phase=" .. tostring(SyncPhase)
                            .. " | state=" .. tostring(SyncState))

                        -- Somente aqui a AUX pode resetar.
                        if SyncRole == "AUX" and not AuxFightDetected then
                            print("[Nuvem V4 SYNC] FIGHT! FINAL detectado na AUX -> campo de batalha confirmado.")
                            print("[Nuvem V4 SYNC] FIGHT! FINAL -> AUX_FIGHT -> reset da AUX.")
                            HandleAuxFight(FightReason)
                        end
                    end
                else
                    -- O banner anterior desapareceu; um próximo FIGHT! poderá
                    -- ser considerado novo.
                    TrialFightWasVisibleAtStart = false
                end
            end

            -- Resultado verde/vermelho só pode ser aceito DEPOIS de FIGHT!.
            -- Isso elimina o falso "TRIAL_COMPLETED" na porta causado por
            -- mensagem antiga ou estado de sincronização.
            if TrialBattleStarted
                and (
                    V4State == "TRIAL_RUNNING"
                    or V4State == "SYNC_WAIT"
                )
            then
                if IsTrialFailureVisible() then
                    HandleTrialFailure()
                elseif IsTrialSuccessVisible() then
                    HandleTrialSuccess()
                end
            end

            -- Libera os detectores somente quando um novo ciclo começa.
            if V4State == "TRIAL_RUNNING" and SyncCycleId ~= nil then
                -- Não limpa aqui enquanto uma mensagem de resultado estiver visível.
                if not TrialFailureReported and not TrialSuccessReported then
                    -- Diagnóstico leve: se a mensagem real tiver outro texto,
                    -- ela aparecerá no console para podermos ajustar sem mexer
                    -- na rota Temple -> porta.
                    if tick() - TrialResultDebugAt >= 2 then
                        TrialResultDebugAt = tick()
                        local ResultTexts = GetTrialResultTexts()
                        for _, ResultText in ipairs(ResultTexts) do
                            if string.find(ResultText, "trial", 1, true)
                                or string.find(ResultText, "complete", 1, true)
                                or string.find(ResultText, "failed", 1, true)
                                or string.find(ResultText, "conclu", 1, true)
                                or string.find(ResultText, "falh", 1, true)
                            then
                                print("[Nuvem V4 RESULT DEBUG] " .. ResultText)
                            end
                        end
                    end
                end
            end
        end)

        task.wait(0.15)
    end
end)

task.spawn(function()

    while Config.Enabled do

        pcall(function()

            local RaidVisible = IsRaidTimerSignalVisible()

            --================================================
            -- PRONTO NA PORTA
            --================================================

            if V4State == "TRIAL"
                and TrialDoorReady
                and not TrialRunning
            then

                if RaidVisible then

                    -- FALLBACK CRÍTICO:
                    -- Se o SyncStart/START_TRIAL não chegar por alguma falha
                    -- momentânea do servidor, o RaidTimer é a confirmação
                    -- local de que a Trial está começando.
                    -- A conta já está na porta, então ativa T agora.
                    if not V3ActivatedForTrial then
                        print("[NUVEM V4] RaidTimer detectado na porta -> ativando V3 localmente.")
                        pcall(function()
                            ActivateV3ForTrial(true)
                        end)
                    end

                    local Race = GetCurrentRace()
                    local Location, _, Trial = GetTrialLocation()

                    TrialRace = Race
                    TrialName = Trial

                    if Location then

                        TrialState = "WAITING_TELEPORT"

                        V4State =
                            "WAITING_TRIAL_TELEPORT"

                        StopMovement()

                        UpdateStatusUI(
                            "TRIAL INICIANDO",
                            "RaidTimer detectado — aguardando teleporte automático...",
                            GetProgressText()
                        )

                        print("==============================================")
                        print("[NUVEM V4] RAIDTIMER DETECTADO")
                        print("[NUVEM V4] Raça:", tostring(Race))
                        print("[NUVEM V4] Trial esperada:", tostring(Trial))
                        print("[NUVEM V4] NÃO haverá teleporte manual para a Trial.")
                        print("[NUVEM V4] Aguardando teleporte automático do jogo...")
                        print("==============================================")

                    else

                        warn(
                            "[NUVEM V4] RaidTimer apareceu, mas a localização da Trial não foi encontrada."
                        )

                    end
                end


            --================================================
            -- ESPERANDO TELEPORTE
            --================================================

            elseif V4State == "WAITING_TRIAL_TELEPORT" then

                SetTrialDoorAnchor(false)

                -- A posição da arena é a confirmação preferencial.
                -- Se a interface não expuser corretamente a Location, o
                -- contador Time Left/Tempo restante também confirma que o
                -- jogo já abriu a Trial. Nenhum teleporte manual é feito aqui.
                if IsPlayerInsideExpectedTrial() or RaidVisible then

                    TrialState = "RUNNING"
                    TrialRunning = true
                    TrialBattleStarted = false

                    local FightAlreadyVisible = IsTrialFightVisible()
                    TrialFightWasVisibleAtStart = FightAlreadyVisible == true

                    print("[Nuvem V4] FIGHT baseline da Trial | visivel_na_inicio="
                        .. tostring(TrialFightWasVisibleAtStart))

                    if TrialFightWasVisibleAtStart then
                        print("[Nuvem V4] FIGHT! já estava visível ao confirmar a Trial -> ignorando até desaparecer e surgir novamente.")
                    end

                    if IsPlayerInsideExpectedTrial() then
                        print("[Nuvem V4] Arena da Trial confirmada pela posição.")
                    else
                        print("[Nuvem V4] Trial confirmada pelo contador Time Left/Tempo restante.")
                    end

                    -- Guarda o estágio ANTES da Trial. É ele que decide
                    -- qual Gear deve ser escolhida quando a Trial terminar.
                    RefreshRealV4Progress()
                    TrialStartProgress = StableV4Progress
                    GearPurchaseDoneForTrial = false

                    V4State = "TRIAL_RUNNING"

                    local Race = GetCurrentRace()
                    local _, _, Trial = GetTrialLocation()

                    TrialRace = Race
                    TrialName = Trial

                    print("==============================================")
                    print("[NUVEM V4] >>> TRIAL CONFIRMADA <<<")
                    print("[NUVEM V4] Raça:", tostring(Race))
                    print("[NUVEM V4] Trial:", tostring(Trial))
                    print("[NUVEM V4] Teleporte automático confirmado.")
                    print("[NUVEM V4] AUTO TRIAL INICIADO.")
                    print("==============================================")

                    UpdateStatusUI(
                        "TRIAL EM ANDAMENTO",
                        "Auto Trial: " .. tostring(Trial),
                        GetProgressText()
                    )

                elseif not RaidVisible then

                    -- O timer sumiu sem o jogo nos colocar na Trial.
                    -- Não força entrada.
                    TrialState = "IDLE"
                    V4State = "CHECKING"

                    print(
                        "[NUVEM V4] RaidTimer desapareceu antes da confirmação da Trial."
                    )

                end


            --================================================
            -- TRIAL RODANDO
            --================================================

            elseif V4State == "TRIAL_RUNNING" then

                SetTrialDoorAnchor(false)

                -- Depois que o jogo confirmou a entrada na Trial, o RaidTimer
                -- passa a ser a fonte principal de verdade durante a luta.
                -- Não usamos mais uma checagem de raio para expulsar a conta:
                -- isso podia interpretar um pequeno deslocamento como saída da
                -- Trial e mandar a conta de volta para a porta no meio da luta.
                if RaidVisible then

                    RunCurrentTrial()

                    UpdateStatusUI(
                        "TRIAL EM ANDAMENTO",
                        "Auto Trial: " .. tostring(TrialName or "detectando..."),
                        GetProgressText()
                    )

                    TrialHiddenSince = nil

                else

                    -- Dá uma pequena margem para evitar falso término
                    -- por um frame de GUI.
                    if not TrialHiddenSince then
                        TrialHiddenSince = tick()
                    end

                    if tick() - TrialHiddenSince >= 1.2 then

                        -- Se o RaidTimer sumiu antes do FIGHT!, a Trial não começou
                        -- de verdade. Volta para o estado de preparação sem marcar
                        -- TRIAL_COMPLETED e sem acionar o fluxo de resultado.
                        if not TrialBattleStarted then
                            print("[Nuvem V4] RaidTimer sumiu antes do FIGHT! -> Trial não iniciada; ignorando falso término.")

                            StopTrialRuntime("RaidTimer removido antes do FIGHT — Trial não iniciada")
                            SyncTrialReported = false
                            SyncWaitingForGroup = false
                            V4State = "TRIAL"
                            TrialDoorReady = true

                            UpdateStatusUI(
                                "PRONTO PARA TRIAL",
                                "Trial não iniciou — aguardando novo início do grupo...",
                                GetProgressText()
                            )

                            TrialHiddenSince = nil
                            return
                        end

                        StopTrialRuntime("RaidTimer removido — aguardando resultado da Trial")
                        ReleaseHumanTrialHitbox()

                        SyncTrialReported = true
                        SyncTrialDoneAt = tick()
                        SyncReport("TRIAL_COMPLETED", "raid_timer_finished_waiting_game_result")

                        -- NÃO compra Gear e NÃO volta para a porta.
                        -- Fica parado até a mensagem verde/vermelha do jogo.
                        V4State = "SYNC_WAIT"
                        SyncWaitingForGroup = true

                        UpdateStatusUI(
                            "AGUARDANDO RESULTADO",
                            "Trial terminou — aguardando confirmação do jogo...",
                            GetProgressText()
                        )

                        print("[NUVEM V4] Trial individual terminou. Personagem parado aguardando mensagem verde/vermelha.")
                    end
                end

            end

        end)

        task.wait(0.1)

    end

end)

    end

--========================================================
-- DIAGNÓSTICO CONTÍNUO
--========================================================

task.spawn(function()

    task.wait(1)

    while Config.Enabled do

        local Ok, Err = xpcall(function()
            CheckV4Status()
        end, function(Error)
            return debug.traceback(tostring(Error), 2)
        end)

        -- Watchdog: se uma exceção inesperada interromper o diagnóstico,
        -- não deixa V4CheckBusy preso para sempre em "Consultando UpgradeRace".
        if not Ok then
            V4CheckBusy = false
            V4State = "UNKNOWN"
            warn("[NUVEM V4] ERRO NO DIAGNÓSTICO; liberando V4CheckBusy:", Err)
            UpdateStatusUI("UNKNOWN", "Erro no diagnóstico — tentando novamente...", GetProgressText())
        end

        task.wait(2)

    end

end)

-- CONTROLE PRINCIPAL
--========================================================

task.spawn(function()

    while Config.Enabled do

        pcall(function()

            if V4State == "TRIAL" then

                -- Depois da Trial, a sincronização é a única autoridade que
                -- pode liberar a auxiliar. Evita que o diagnóstico do Ancient
                -- One reabra a rota sozinho.
                if SyncRole == "AUX"
                    and SyncPostResetRoute
                then
                    StopMovement()
                    StopTrialDoorMovement()
                    TrialDoorReady = false
                    V4State = "SYNC_WAIT"

                    UpdateStatusUI(
                        "AGUARDANDO GRUPO",
                        "Reset concluído — aguardando liberação das 3 contas...",
                        GetProgressText()
                    )

                    return
                end

                CurrentTarget = nil

                if not TempleReady then

                    StopMovement()

                    TrialDoorReady = false

                    UpdateStatusUI(
                        "PRONTO PARA TRIAL",
                        "Indo para o Temple of Time...",
                        GetProgressText()
                    )

                    task.spawn(
                        GoToTempleForTrial
                    )

                elseif not TrialDoorReady then

                    if AncientOneProbeRequired
                        and not AncientOneProbeDone
                        and not Config.DirectTempleToTrial
                    then
                        local ProbeFinished = RunAncientOneProbe()

                        if V4State == "TRAINING" then
                            -- O diálogo informou que ainda precisa treinar.
                            -- Não manda a conta para a porta.
                            return
                        end

                        if not ProbeFinished then
                            return
                        end
                    end

                    print(
                        "[Nuvem V4] TempleReady=true | tentando ir para a porta da Trial."
                    )

                    GoToTrialDoor()

                else

                    StopMovement()

                    local Root = GetRoot()

                    local TargetCFrame =
                        GetTrialDoorCFrame()

                    -- IMPORTANTE: não reaplica CFrame continuamente na porta.
                    -- Depois que chegou, a conta permanece parada e o servidor
                    -- sincroniza o grupo. Isso evita o loop porta/Temple.
                    if Root then
                        Root.AssemblyLinearVelocity = Vector3.zero
                        Root.AssemblyAngularVelocity = Vector3.zero
                    end

                    UpdateStatusUI(
                        "PRONTO PARA TRIAL",
                        "Na porta da Trial — aguardando",
                        GetProgressText()
                    )

                end

            elseif V4State == "WAITING_TRIAL_TELEPORT" then

                -- Não mexe no personagem.
                -- O jogo é responsável pelo teleporte da Trial.
                StopMovement()

                UpdateStatusUI(
                    "TRIAL INICIANDO",
                    "Aguardando teleporte automático...",
                    GetProgressText()
                )

            elseif V4State == "TRIAL_RUNNING" then

                -- O monitor da Trial controla movimento/ataque.
                -- O loop principal não tenta voltar para a porta.
                TrialDoorReady = false

            elseif V4State == "SYNC_WAIT" then

                StopTrialDoorMovement()
                StopMovement()
                CurrentTarget = nil

                UpdateStatusUI(
                    "AGUARDANDO GRUPO",
                    SyncRole == "MAIN" and "Aguardando as auxiliares..." or "Trial concluída — aguardando o grupo...",
                    GetProgressText()
                )

            elseif V4State == "COMPLETE" then

                SetTrialDoorAnchor(false)

                TrialDoorReady = false
                StopTrialDoorMovement()

                StopMovement()

                CurrentTarget = nil

                UpdateStatusUI(
                    "RAÇA COMPLETA",
                    "Nenhuma ação necessária",
                    "10/10"
                )

            elseif V4State == "TRAINING" then

                SetTrialDoorAnchor(false)

                StopTrialDoorMovement()

                UpdateStatusUI(
                    "PRECISA TREINAR",
                    "Treinando em Bones",
                    GetProgressText()
                )

            elseif V4State == "BUY_AVAILABLE" then

                SetTrialDoorAnchor(false)

                StopTrialDoorMovement()

                UpdateStatusUI(
                    "COMPRA DISPONÍVEL",
                    "Pagando treinamento...",
                    GetProgressText()
                        .. " - $"
                        .. tostring(V4Price)
                )

            elseif V4State == "GEAR_WAITING" then

                StopTrialDoorMovement()

                UpdateStatusUI(
                    "AGUARDANDO GEAR",
                    "Trial concluída — aguardando compra da Gear...",
                    GetProgressText()
                )

            elseif V4State == "UNKNOWN" then

                StopTrialDoorMovement()

                UpdateStatusUI(
                    "UNKNOWN",
                    "Consultando Ancient One...",
                    "--"
                )

            else

                UpdateStatusUI(
                    "VERIFICANDO",
                    "Consultando UpgradeRace...",
                    "--"
                )

            end

        end)

        task.wait(0.5)

    end

end)
